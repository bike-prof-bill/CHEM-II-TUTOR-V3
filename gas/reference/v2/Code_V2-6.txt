// --- CONFIGURATION ---
// Primary and secondary fallback models
var PRIMARY_MODEL = "gemini-3.6-flash";
var FALLBACK_MODEL = "gemini-3.5-flash";

// V2. Written into every Log row, so a rule change can be dated against a row
// without anyone's memory. Change it on every deploy.
var BUILD_STAMP = "V2-2026-09-18-d";

// --- HELPER FUNCTIONS ---

// A short fingerprint of the problem a session is working on: the archetype key
// plus a hash of the problem text. Two problems of the same archetype get
// different fingerprints.
// One place where an anonymous student is named. Two defaults in two branches
// meant the Log wrote "Guest (No Extra Credit)" while Mastery wrote "Guest", and
// the sheets would not join on anonymous rows.
function studentOf(requestData) {
  return (requestData && requestData.studentId) || "Guest (No Extra Credit)";
}

function problemId(prob) {
  if (!prob) return "none";
  var key = String(prob.key || "custom");
  var s = key + "|" + String(prob.text || "");
  var h = 5381;
  for (var i = 0; i < s.length; i++) { h = ((h * 33) ^ s.charCodeAt(i)) >>> 0; }
  return key + ":" + h.toString(36);
}

// Returns the saved milestone for this session, but only if the saved problem
// matches the one being sent now. A session whose problem has been swapped
// starts again at 0.
function getMilestone(sessionId, probId) {
  if (!sessionId) return 0;
  let cache = CacheService.getScriptCache();
  let cacheKey = "ms_" + sessionId;
  let cached = cache.get(cacheKey);
  if (cached !== null) {
    let sep = cached.lastIndexOf("|");
    if (sep > 0) {
      let savedId = cached.slice(0, sep);
      let savedMs = parseInt(cached.slice(sep + 1), 10) || 0;
      return sameProblem(savedId, probId) ? savedMs : 0;
    }
    // older cache format (a bare number): fall through to the sheet
  }

  const SHEET_ID = PropertiesService.getScriptProperties().getProperty("SHEET_ID");
  let sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName("State");
  let rowNum = findSessionRow(sheet, sessionId);

  let savedId = "";
  let savedMs = 0;
  if (rowNum) {
    // One read for the whole row instead of a round trip per cell.
    let row = sheet.getRange(rowNum, 1, 1, 4).getValues()[0];
    savedId = String(row[2]);
    savedMs = parseInt(row[3], 10) || 0;
  }

  // ALWAYS cache, including "this session is not in the sheet yet". Caching
  // only on a hit meant that every message of a new session re-scanned the
  // State sheet, which is what made the tutor crawl.
  cache.put(cacheKey, savedId + "|" + savedMs, 21600);
  return sameProblem(savedId, probId) ? savedMs : 0;
}

// Bounded lookup. createTextFinder over "A:A" searches the entire column; the
// used range is a few hundred rows and comes back in a single read.
function findSessionRow(sheet, sessionId) {
  let last = sheet.getLastRow();
  if (last < 1) return 0;
  let ids = sheet.getRange(1, 1, last, 1).getValues();
  for (let i = ids.length - 1; i >= 0; i--) {
    if (String(ids[i][0]) === sessionId) return i + 1;
  }
  return 0;
}

// Rows saved before this change hold only the archetype key ("ch11_henry").
// Trust those on the key alone so nobody mid-problem is sent back to 0.
function sameProblem(savedId, probId) {
  if (savedId === probId) return true;
  if (savedId.indexOf(":") < 0) return savedId === probId.split(":")[0];
  return false;
}

function setMilestone(sessionId, studentId, probId, n) {
  if (!sessionId) return;
  let cache = CacheService.getScriptCache();
  let cacheKey = "ms_" + sessionId;
  cache.put(cacheKey, probId + "|" + n, 21600);

  const SHEET_ID = PropertiesService.getScriptProperties().getProperty("SHEET_ID");
  let lock = LockService.getScriptLock();
  if (lock.tryLock(3000)) {
    try {
      let sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName("State");
      let row = findSessionRow(sheet, sessionId);
      if (row) {
        sheet.getRange(row, 2, 1, 4).setValues([[studentId, probId, n, new Date()]]);
      } else {
        sheet.appendRow([sessionId, studentId, probId, n, new Date()]);
      }
    } finally {
      lock.releaseLock();
    }
  }
}

// One attempt = one student working one problem from start to finish or until
// they leave it. V1 left this to be inferred from the milestone going DOWN,
// which is invisible when a student clicks New Scenario and never types again.
// Kept in the cache for 6 hours: a student idle longer than that on the same
// problem gets a new attempt ID for the rest of it.
function attemptFor(sessionId, probId) {
  if (!sessionId) return "";
  var cache = CacheService.getScriptCache();
  var raw = cache.get("att_" + sessionId);
  if (raw) {
    var sep = raw.lastIndexOf("|");
    if (sep > 0 && raw.slice(0, sep) === probId) return raw.slice(sep + 1);
  }
  var id = "A" + Date.now().toString(36) + Math.floor(Math.random() * 46656).toString(36);
  cache.put("att_" + sessionId, probId + "|" + id, 21600);
  return id;
}

function endAttempt(sessionId) {
  if (!sessionId) return;
  CacheService.getScriptCache().put("att_" + sessionId, "ended|0", 21600);
  CacheService.getScriptCache().put("last_" + sessionId, "{}", 300);
}

// --- ROUTER ---
function doPost(e) {
  // Column 20: how long the script itself held the request, start to log write.
  // Column 11 is the model alone. A student's stopwatch minus column 20 is time
  // spent outside the script — Google's redirect hop and start-up.
  var serverT0 = Date.now();
  try {
    const props = PropertiesService.getScriptProperties();
    const APP_TOKEN = props.getProperty("APP_TOKEN");
    const SHEET_ID = props.getProperty("SHEET_ID");

    let requestData = JSON.parse(e.postData.contents);

    if (requestData.authToken !== APP_TOKEN) {
      return createJsonResponse({ error: "unauthorized" });
    }

    let sessionId = requestData.sessionId;
    let probId = problemId(requestData.problemConfig);

    // A POST to /exec redirects to a one-shot script.googleusercontent.com URL,
    // and that second hop intermittently dies — the student sees HTTP 404 or
    // Safari's "Load failed". The page now retries. But a retry is only safe if
    // the server cannot process the same turn twice: the first attempt may well
    // have run to completion and merely lost its reply on the way back. So every
    // send carries a turnId, and a repeat of one already answered returns the
    // stored reply instead of grading, logging and billing the turn again.
    if (requestData.turnId) {
      var seen = CacheService.getScriptCache().get("turn_" + requestData.turnId);
      if (seen) return ContentService.createTextOutput(seen)
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Static text, no model call: the page asks for this on load, so it must be
    // fast and must never cost a round trip to Gemini.
    if (requestData.action === "opener") {
      return createJsonResponse({ opener: openerFor(requestData.problemConfig) });
    }

    if (requestData.action === "reset") {
      let studentId = studentOf(requestData);
      // A student clicking New Scenario part-way through has abandoned that
      // problem. Without this the Mastery sheet only ever shows successes, and
      // where students give up is the more useful half of the record.
      try {
        var leaving = getMilestone(sessionId, probId);
        if (leaving > 0 && leaving < 5) {
          recordMastery(studentId, requestData.problemConfig,
                        getScaffold(sessionId, probId), leaving);
        }
      } catch (e) {}
      setMilestone(sessionId, studentId, probId, 0);
      endAttempt(sessionId);
      return createJsonResponse({ status: "reset" });
    }

    if (sessionId) {
      let cache = CacheService.getScriptCache();
      let cacheKey = "rate_" + sessionId;
      let currentCount = parseInt(cache.get(cacheKey) || "0", 10);
      if (currentCount >= 40) {
        return createJsonResponse({ error: "rate limit reached, please wait" });
      }
      cache.put(cacheKey, (currentCount + 1).toString(), 3600);
    }

    let history = requestData.history || [];
    history = history.slice(-10);

    let currentMilestone = getMilestone(sessionId, probId);

    // The turnId guard covers the page's own automatic retry. It does not cover
    // the student pressing Send again after both attempts failed: that is a new
    // turnId. The first attempt usually DID run — it was graded, logged, and its
    // reply lost on the way back. (V2 live run, 16 Sept: the lost reply was the
    // good question, the retry got a worse one, and the rung rose a step for a
    // network failure.) Same session, same problem, same words, nothing changed
    // since: hand back the reply they never saw.
    if (sessionId) {
      try {
        var lastRaw = CacheService.getScriptCache().get("last_" + sessionId);
        if (lastRaw) {
          var last = JSON.parse(lastRaw);
          var sameWords = String(last.msg).trim() === String(requestData.message || "").trim();
          if (sameWords && last.probId === probId && last.reply && last.reply.milestone === currentMilestone) {
            return ContentService.createTextOutput(JSON.stringify(last.reply))
              .setMimeType(ContentService.MimeType.JSON);
          }
        }
      } catch (e) {}
    }

    // --- STEP 3: grade the numbers before the tutor speaks ---
    let workingMilestone = currentMilestone + 1;
    let target = targetsFor(requestData.problemConfig, workingMilestone);
    let grade = gradeMessage(requestData.message, target, requestData.problemConfig);
    let recent = recentStudentText(history, requestData.message);
    // A step that needs two values (two temperatures) may get them one per turn.
    // The grader reads only this message, so without this the tutor is told
    // "partial, do not clear" on the turn the second value arrives.
    if (target && target.all && grade.verdict === "PARTIAL" &&
        satisfiesAnywhere(recent, target, requestData.problemConfig, [])) {
      grade = { verdict: "CORRECT", label: target.label, student: "across their last messages" };
    }
    // A student who answers in two parts ("0.085", then "M" when asked for
    // units) has still answered. Chaining looks at the last few student turns,
    // while the turn-by-turn verdict above still reads only this message, so
    // traps land exactly where they did before.
    let scaffold = getScaffold(sessionId, probId);
    let verdictBlock = buildVerdictBlock(grade, requestData.problemConfig, requestData.message, currentMilestone, recent) + signNote(requestData.problemConfig, workingMilestone, grade)
      + trapWatch(requestData.problemConfig)
      + freshnessNote(history);

    // The rung is computed from the PREVIOUS turns, so the tutor is told how much
    // help to give before it answers. Only the one rung that applies is injected.
    verdictBlock += buildScaffoldNote(scaffold, "answer");

    // An equation the student typed, checked by substitution. Only once the
    // problem is visible (steps 2 to 4). A failed check blocks the step; an
    // unreadable one blocks it only when nothing else in the message verifies.
    let eq = (currentMilestone >= 1 && currentMilestone <= 3)
      ? checkEquation(requestData.message, requestData.problemConfig) : null;
    let numberVerified = grade && (grade.verdict === "CORRECT" || grade.verdict === "CORRECT_WITH_STRAY");
    // Unreadable blocks only when the brackets do not pair: that is a real error and
    // is reliably seen. Any other unreadable input may be the checker's own failure,
    // and a correct "Ecell = ... = 0.52 V." was blocked that way (17 Sept).
    let eqBlocks = !!eq && eq.verdict !== "VALID" && !(eq.verdict === "UNREADABLE" && (eq.why !== "paren" || numberVerified));
    verdictBlock += equationBlock(eq, requestData.problemConfig);
    verdictBlock += caretNote(requestData.message);
    verdictBlock += reflectionNote(scaffold, currentMilestone);

    // How far this message would carry the student if the tutor accepts the step
    // it is judging. Computed before the model call, so the prompt can show the
    // one step the tutor will need next and nothing beyond it.
    let reachIfCleared = eqBlocks ? 0 : grantMilestones(requestData.problemConfig, recent, currentMilestone,
                                         5 - currentMilestone, false, requestData.message);

    let t0 = Date.now();
    let called = callGeminiAPIWithFallback(requestData.message, history, requestData.problemConfig, currentMilestone, verdictBlock, reachIfCleared);
    let aiResponse = called.reply;
    let modelUsed = called.model;
    var modelMs = Date.now() - t0;

    let claimed = claimedFromResponse(aiResponse);
    let rawClaim = claimed;

    // A second, independent reading of a step only the tutor can judge. When the
    // tutor is about to clear a step with no numeric answer key (the particle
    // account, the look-back, words-only steps), a separate call reads the
    // student's message for a false statement about the chemistry. If it finds one
    // — and the phrase it quotes really is in the student's message — the step is
    // not cleared and the reply is rewritten to take up the error instead of
    // endorsing it. Simulated runs, 17 Sept: "vinegar molecules have much stronger
    // bonds so water can't break them" was answered "yes"; "energy that stays
    // stored inside the nucleus" was answered "you correctly identified that the
    // mass was converted into energy released".
    // Milestone 1 only: an account that touches nothing in this scenario does not
    // clear, however well it is phrased. Stands down from scaffold rung 2, so a
    // student who has already been asked twice is never held by vocabulary.
    let scenarioGate = "";
    if (claimed >= 1 && currentMilestone === 0 && scaffold.rung <= 1) {
      let cov = openerCoverage(recent + " " + requestData.message, requestData.problemConfig);
      // Refused only when the message touches NOTHING in the scenario AND says
      // nothing mechanistic of its own: "it is at dynamic equilibrium", "Le
      // Chatelier", a bare law name. Measured against 102 real milestone-1 clears
      // from the simulated runs, this refuses none of them; the instructor's own
      // model answers touch the scenario 2 to 6 times.
      let mech = gateMechanismWords(requestData.message);
      if (!cov.ungated && cov.hits === 0 && mech < 2) {
        scenarioGate = "OPENER_NOT_ENGAGED";
        claimed = 0;
        try {
          let ts = Date.now();
          let redo = callGeminiAPIWithFallback(requestData.message, history, requestData.problemConfig, currentMilestone,
            verdictBlock + gateNote(cov), reachIfCleared);
          modelMs += Date.now() - ts;
          if (redo.reply && redo.reply.socratic_response) aiResponse.socratic_response = redo.reply.socratic_response;
        } catch (e) {}
      }
    }

    let errorGuard = "";
    let wordsOnlyStep = !target || (grade && grade.verdict === "NO_KEY");
    if (claimed >= 1 && wordsOnlyStep && !eqBlocks && currentMilestone <= 4) {
      try {
        let tj = Date.now();
        let found = findStudentError(requestData.message, requestData.problemConfig, currentMilestone + 1, aiResponse.socratic_response);
        modelMs += Date.now() - tj;
        if (found) {
          errorGuard = "REPLY_AFFIRMED_ERROR";
          claimed = 0;
          let tr = Date.now();
          let redo = callGeminiAPIWithFallback(requestData.message, history, requestData.problemConfig, currentMilestone,
            verdictBlock + errorNote(found, aiResponse.socratic_response), reachIfCleared);
          modelMs += Date.now() - tr;
          if (redo.reply && redo.reply.socratic_response) {
            aiResponse.socratic_response = redo.reply.socratic_response;
            aiResponse.turn_kind = redo.reply.turn_kind || aiResponse.turn_kind;
          }
        }
      } catch (e) {}
    }

    // V2: the tutor only sees the current step, so it can only say yes (1) or no
    // (0) to that step. When it says yes, the numbers decide how much further the
    // student has already gone — grantMilestones applies every cap it always did.
    let claimForGrant = eqBlocks ? 0 : ((claimed >= 1) ? Math.max(claimed, 5 - currentMilestone) : claimed);
    let granted = grantMilestones(requestData.problemConfig, recent, currentMilestone, claimForGrant, false, requestData.message);
    let isCleared = granted > 0;
    let turnKind = String(aiResponse.turn_kind || "answer");
    if (turnKind !== "question" && turnKind !== "offtopic" && turnKind !== "reflection") turnKind = "answer";
    let nextSc = nextScaffold(scaffold, isCleared, turnKind);
    setScaffold(sessionId, probId, nextSc);
    let rawMilestoneCleared = rawClaim;
    let newMilestone = currentMilestone + granted;

    // Rule 2 enforced by the server: a reply may not state a value from the answer
    // key that the student has not produced. The simulated students caught it three
    // times in 120 turns, always the same way ("x = 7.6e-5 M, which yields a pH of
    // 4.12"). The draft is sent back once with the value named; the milestones
    // already granted stand.
    let studentText = history.filter(function (t) { return t.role === "user"; })
                             .map(function (t) { return t.text; }).join(" ") + " " + (requestData.message || "");
    let leaked = unearnedValues(aiResponse.socratic_response, requestData.problemConfig, currentMilestone, studentText);
    let handed = handedSetup(aiResponse.socratic_response, requestData.problemConfig, currentMilestone, studentText);
    let leakGuard = "";
    if (leaked.length || handed) {
      leakGuard = leaked.length ? "REPLY_STATED_VALUE" : "REPLY_GAVE_SETUP";
      try {
        let t1 = Date.now();
        let redo = callGeminiAPIWithFallback(requestData.message, history, requestData.problemConfig, currentMilestone,
          verdictBlock + leakNote(leaked, aiResponse.socratic_response, handed), reachIfCleared);
        modelMs += Date.now() - t1;
        var r2 = redo.reply && redo.reply.socratic_response;
        var better = r2 &&
          unearnedValues(r2, requestData.problemConfig, currentMilestone, studentText).length <= leaked.length &&
          (!handed || !handedSetup(r2, requestData.problemConfig, currentMilestone, studentText) ||
           unearnedValues(r2, requestData.problemConfig, currentMilestone, studentText).length < leaked.length);
        if (better) aiResponse.socratic_response = r2;
      } catch (e) {}
    }

    // One row per finished problem. Silent: the student sees nothing change.
    if (newMilestone >= 5 && currentMilestone < 5) {
      // studentId, not requestData.studentId — otherwise Log says
      // "Guest (No Extra Credit)" and Mastery says "Guest", and the two sheets
      // will not join on anonymous rows.
      recordMastery(studentOf(requestData), requestData.problemConfig, nextSc, newMilestone);
    }

    let studentId = studentOf(requestData);
    if (isCleared && currentMilestone < 5) {
      setMilestone(sessionId, studentId, probId, newMilestone);
    } else if (sessionId) {
      // Nothing cleared, but remember which problem this session is on so the
      // next message need not go back to the sheet to find out.
      CacheService.getScriptCache().put("ms_" + sessionId, probId + "|" + newMilestone, 21600);
    }

    let lock = LockService.getScriptLock();
    if (lock.tryLock(3000)) {
      try {
        SpreadsheetApp.openById(SHEET_ID).getSheetByName("Log").appendRow([
          new Date(),
          sessionId,
          studentId,
          requestData.chapter || "?",
          requestData.problemName || "?",
          requestData.message || "?",
          aiResponse.socratic_response,
          isCleared,
          newMilestone,
          rawMilestoneCleared,
          modelMs,
          modelUsed,
          scaffold.rung,
          turnKind,
          probId,
          (grade && grade.trap) ? grade.trap.id : (scenarioGate || errorGuard || equationTrapId(eq) || leakGuard),
          currentMilestone,
          attemptFor(sessionId, probId),
          BUILD_STAMP,
          Date.now() - serverT0,
          // Column 21: every guard that fired this turn. Column 16 holds one token by
          // precedence, so a guard on a turn that also had a grader trap or an equation
          // error left no trace — and those are exactly the turns where the tutor is
          // most likely to reach for the setup or state a value. (Analysis side,
          // 18 Sept: the undercount was biased toward zero in the population the
          // measurement exists to examine.)
          [scenarioGate, errorGuard, leakGuard].filter(String).join(",")
        ]);
      } finally {
        lock.releaseLock();
      }
    }

    // Store the reply against this turnId so a retry replays it rather than
    // grading, logging and billing the same message a second time.
    var replyJson = JSON.stringify({
      socratic_response: aiResponse.socratic_response,
      milestone_cleared: isCleared,
      milestone: newMilestone
    });
    if (sessionId) {
      try {
        CacheService.getScriptCache().put("last_" + sessionId, JSON.stringify({
          msg: requestData.message || "", probId: probId, reply: JSON.parse(replyJson) }), 300);
      } catch (e) {}
    }
    if (requestData.turnId) {
      try { CacheService.getScriptCache().put("turn_" + requestData.turnId, replyJson, 600); } catch (e) {}
    }
    return ContentService.createTextOutput(replyJson).setMimeType(ContentService.MimeType.JSON);
  } catch (error) { return createJsonResponse({ error: error.toString() }); }
}

// The solve block is the grader's answer key and also carries constants the
// student is supposed to look up (Kf, Kb, molar masses). The model never needed
// it and was reading values straight out of it into its questions, so it stays
// on the server. Only the cosmetic fields the on-screen simulations use are
// passed through.
function displayVars(prob) {
  if (!prob || !prob.vars) return {};
  var out = {};
  for (var k in prob.vars) {
    if (Object.prototype.hasOwnProperty.call(prob.vars, k) && k !== "solve") out[k] = prob.vars[k];
  }
  return out;
}

// The five step descriptions for an archetype, as an array indexed 1..5.
function stepLines(key) {
  var w = String(WORKFLOWS[key] || DEFAULT_WORKFLOW).replace(/^MILESTONES:\s*/, "");
  var parts = w.split(/\n(?=[1-5]\. )/);
  var out = ["", "", "", "", "", ""];
  for (var i = 0; i < parts.length; i++) {
    var m = parts[i].match(/^([1-5])\.\s*([\s\S]*)$/);
    if (m) out[parseInt(m[1], 10)] = m[2].trim();
  }
  return out;
}

// Returns { reply, model }. Which model answered goes into the Log sheet: if
// the primary is failing, every request silently pays for two round trips, and
// that shows up as FALLBACK in the log rather than as a mystery slowdown.
function callGeminiAPIWithFallback(message, history, problemConfig, currentMilestone, verdictBlock, reach) {
  try {
    return { reply: executeGeminiCall(message, history, problemConfig, PRIMARY_MODEL, currentMilestone, verdictBlock, reach),
             model: PRIMARY_MODEL };
  } catch (e) {
    return { reply: executeGeminiCall(message, history, problemConfig, FALLBACK_MODEL, currentMilestone, verdictBlock, reach),
             model: FALLBACK_MODEL + " after " + String(e).slice(0, 40) };
  }
}

function executeGeminiCall(message, history, problemConfig, modelName, currentMilestone, verdictBlock, reach)  {
  const GEMINI_API_KEY = PropertiesService.getScriptProperties().getProperty("GEMINI_API_KEY");
  let url = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${GEMINI_API_KEY}`;
  let contents = history.map(t => ({ role: t.role, parts: [{ text: t.text }] }));
  contents.push({ role: "user", parts: [{ text: message }] });

  let payload = {
    system_instruction: { parts: [{ text: buildSystemPrompt(problemConfig, currentMilestone, verdictBlock, reach) }] },
    contents: contents,
    generationConfig: { response_mime_type: "application/json" }
  };

  let response = UrlFetchApp.fetch(url, { method: "post", contentType: "application/json", payload: JSON.stringify(payload), muteHttpExceptions: true });
  if (response.getResponseCode() === 200) {
    return JSON.parse(JSON.parse(response.getContentText()).candidates[0].content.parts[0].text);
  } else throw new Error(`API Error ${response.getResponseCode()}`);
}

// =============================================================================
// EQUATION CHECKER (V2, build h)
// =============================================================================
//
// The numeric grader reads numbers. A rearranged equation typed as symbols has
// none, so until now only the model judged it — and log-and-reciprocal
// rearrangements are where students (and models) flip signs. This checks an
// equation the student SUBMITS by substituting the problem's own values,
// including the correct answer, into both sides. Every valid rearrangement
// passes without being listed; a flipped one fails, and the kind of flip is
// named. Nothing here looks ahead: it judges only what was typed.
//
// Symbol names follow the Equations list. Anything it cannot read is reported as
// unreadable, never guessed at.

var EQ_FUNCS = { ln: Math.log, log: Math.log10, exp: Math.exp, sqrt: Math.sqrt };

// Returns { env, unknown, flips, names, format } or null for an archetype the
// checker does not cover. `flips` are the wrong-but-common values of the
// unknown: if the equation holds for one of those, that is the diagnosis.
function eqSpec(prob) {
  var s = prob && prob.vars && prob.vars.solve;
  if (!s) return null;
  var R = R_J, env, unknown, flips = [], fmt;
  if (s.type === "arrhenius") {
    var T1 = s.t1C + 273.15, T2 = s.t2C + 273.15;
    env = { k1: s.k1, T1: T1, T2: T2, R: R };
    if (s.solveFor === "ea") {
      env.k2 = s.k2;
      env.Ea = R * Math.log(s.k2 / s.k1) / (1 / T1 - 1 / T2);
      unknown = "Ea";
      flips.push({ id: "EQ_SIGN_FLIPPED", value: -env.Ea });
    } else {
      env.Ea = s.ea * 1000;
      env.k2 = s.k1 * Math.exp(-(env.Ea / R) * (1 / T2 - 1 / T1));
      unknown = "k2";
      flips.push({ id: "EQ_SIGN_FLIPPED", value: s.k1 * s.k1 / env.k2 });
    }
    fmt = "k1, k2, T1, T2, Ea, R";
  } else if (s.type === "cc") {
    var cT1 = s.tC + 273.15;
    env = { P1: s.p1, P2: s.p2, T1: cT1, R: R, dH: s.dh * 1000 };
    env.T2 = 1 / (1 / cT1 - (R / env.dH) * Math.log(s.p2 / s.p1));
    unknown = "T2";
    flips.push({ id: "EQ_SIGN_FLIPPED", value: 1 / (1 / cT1 + (R / env.dH) * Math.log(s.p2 / s.p1)) });
    fmt = "P1, P2, T1, T2, dHvap, R";
  } else if (s.type === "integrated") {
    env = { A0: s.a0, k: s.k, t: s.t };
    if (s.order === 1) {
      env.At = s.a0 * Math.exp(-s.k * s.t);
      flips.push({ id: "EQ_SIGN_FLIPPED", value: s.a0 * Math.exp(s.k * s.t) });
    } else if (s.order === 2) {
      env.At = 1 / (1 / s.a0 + s.k * s.t);
      flips.push({ id: "EQ_RECIPROCAL_MISSING", value: 1 / env.At });
      flips.push({ id: "EQ_SIGN_FLIPPED", value: 1 / (1 / s.a0 - s.k * s.t) });
    } else {
      env.At = s.a0 - s.k * s.t;
      flips.push({ id: "EQ_SIGN_FLIPPED", value: s.a0 + s.k * s.t });
    }
    unknown = "At";
    fmt = "[A]0, [A]t, k, t";
  } else if (s.type === "keq") {
    var kT = s.T || 298;
    env = { R: R, T: kT };
    if (s.mode === "findG") {
      env.K = s.K; env.dG0 = -R * kT * Math.log(s.K); unknown = "dG0";
      flips.push({ id: "EQ_SIGN_FLIPPED", value: -env.dG0 });
    } else {
      env.dG0 = s.dg0 * 1000; env.K = Math.exp(-env.dG0 / (R * kT)); unknown = "K";
      flips.push({ id: "EQ_SIGN_FLIPPED", value: 1 / env.K });
    }
    fmt = "dG°, R, T, K";
  } else if (s.type === "nonstandard") {
    var nT = s.T || 298;
    env = { R: R, T: nT, Q: eqProd(s.num) / eqProd(s.den), dG0: s.dg0 * 1000 };
    env.dG = env.dG0 + R * nT * Math.log(env.Q);
    unknown = "dG";
    flips.push({ id: "EQ_SIGN_FLIPPED", value: env.dG0 - R * nT * Math.log(env.Q) });
    fmt = "dG, dG°, R, T, Q";
  } else if (s.type === "nernst") {
    var Q = eqProd(s.num) / eqProd(s.den);
    env = { E0: s.e0, n: s.n, Q: Q, R: R, T: 298.15, F: 96485 };
    env.E = s.e0 - (0.0592 / s.n) * Math.log10(Q);
    unknown = "E";
    flips.push({ id: "EQ_SIGN_FLIPPED", value: s.e0 + (0.0592 / s.n) * Math.log10(Q) });
    flips.push({ id: "EQ_LOG_BASE", value: s.e0 - (0.0592 / s.n) * Math.log(Q) });
    fmt = "E, E°, n, Q";
  } else {
    return null;
  }
  var names = Object.keys(env).concat(["e"]);
  return { env: env, unknown: unknown, flips: flips, names: names, format: fmt, type: s.type };
}

function eqProd(list) {
  var p = 1;
  for (var i = 0; i < (list || []).length; i++) p *= Math.pow(list[i][0], list[i][1]);
  return p;
}

var EQ_SUP = { "˙":".", "⁰":"0","¹":"1","²":"2","³":"3","⁴":"4","⁵":"5","⁶":"6","⁷":"7","⁸":"8","⁹":"9","⁻":"-","⁺":"+" };

// ---- units -----------------------------------------------------------------
// A number may carry units. They are read, not stripped: the value is used as
// typed, and the units are carried through the arithmetic so the checker can
// say whether they cancel. Raw units are what the student wrote; SI units are
// the same after conversion (kJ to J, min to s, M to mol/L, V to J/C, mmHg and
// torr to atm). Raw agreeing means the units cancel as written; only SI agreeing
// means they cancel after a conversion the student has not made.
var EQ_UNIT_WORDS = ["degC", "mmHg", "torr", "atm", "mol", "min", "kJ", "J", "K", "s", "M", "L", "V", "C"];
var EQ_UNIT_SI = {
  kJ: { J: 1 }, J: { J: 1 }, mol: { mol: 1 }, K: { K: 1 }, degC: { degC: 1 }, s: { s: 1 }, min: { s: 1 },
  M: { mol: 1, L: -1 }, L: { L: 1 }, V: { J: 1, C: -1 }, C: { C: 1 }, atm: { atm: 1 }, mmHg: { atm: 1 }, torr: { atm: 1 }
};
var EQ_UNIT_FACTOR = { kJ: 1000, min: 60, mmHg: 1 / 760, torr: 1 / 760 };   // to SI
// What each named symbol means, in SI. null = depends on the problem (a rate
// constant's units follow the order; a concentration may be a pressure).
var EQ_SYMBOL_SI = {
  Ea: { J: 1, mol: -1 }, dH: { J: 1, mol: -1 }, dG: { J: 1, mol: -1 }, dG0: { J: 1, mol: -1 },
  R: { J: 1, mol: -1, K: -1 }, T: { K: 1 }, T1: { K: 1 }, T2: { K: 1 }, t: { s: 1 },
  E: { J: 1, C: -1 }, E0: { J: 1, C: -1 }, F: { C: 1, mol: -1 }, n: {}, Q: {}, K: {}, e: {},
  k: null, k1: null, k2: null, A0: null, At: null, P1: null, P2: null
};

function eqVecAdd(a, b, sign) {
  var out = {}, k;
  for (k in a) out[k] = (out[k] || 0) + a[k];
  for (k in b) out[k] = (out[k] || 0) + sign * b[k];
  for (k in out) if (Math.abs(out[k]) < 1e-9) delete out[k];
  return out;
}
function eqVecScale(a, f) { var out = {}; for (var k in a) out[k] = a[k] * f; return eqVecAdd(out, {}, 1); }
function eqVecEmpty(a) { return Object.keys(a).length === 0; }
function eqVecSame(a, b) { return eqVecEmpty(eqVecAdd(a, b, -1)); }

// Plain-text cleanup. Spaces are kept, because units are separated by them.
function eqNormalize(text, type) {
  var s = String(text || "")
    .replace(/[\u200b\u200c\u200d\ufeff]/g, "")
    .replace(/[\u00a0\t]/g, " ")
    .replace(/[₀₁₂₃₄₅₆₇₈₉]/g, function (c) { return String(c.charCodeAt(0) - 0x2080); })
    .replace(/(\d),(\d{3})(?![\d.])/g, "$1$2")
    .replace(/[⁺⁻]?[⁰¹²³⁴⁵⁶⁷⁸⁹][⁰¹²³⁴⁵⁶⁷⁸⁹˙]*/g, function (m) {
      return "^(" + m.split("").map(function (c) { return EQ_SUP[c]; }).join("") + ")";
    })
    .replace(/[×·⋅∙•]/g, "*").replace(/[−–—]/g, "-").replace(/÷/g, "/")
    .replace(/[°º]\s*C(?![a-z])/g, " degC")
    .replace(/Δ/g, "d").replace(/[°º]/g, "0");
  // 1.50 × 10⁻² (the toolbar's "× 10" and superscripts) becomes 1.50e-2
  s = s.replace(/(\d*\.?\d+)\s*[x*]\s*10\s*\^\s*\(?\s*([-+]?\d+)\s*\)?/g, "$1e$2");
  if (type === "integrated") {
    s = s.replace(/\[[^\]]*\]\s*_?\s*(?:0|o|i)(?![A-Za-z0-9])/g, "A0")
         .replace(/\[[^\]]*\]\s*_?\s*t(?![A-Za-z0-9])/g, "At")
         .replace(/\[[^\]]*\]/g, "At")
         .replace(/\bP_?(?:0|o|i)\b/g, "A0").replace(/\bP_?t\b/g, "At")
         .replace(/\bA_?(?:o|i)\b/g, "A0");
  }
  s = s.replace(/_/g, "")
       .replace(/\bE\s+a\b/g, "Ea").replace(/\b([kTP])\s+([0-2])\b/g, "$1$2")
       .replace(/E0\s*cell|E0cel/g, "E0").replace(/E\s*cell/g, "E")
       .replace(/dH\s*vap|H\s*vap/g, "dH").replace(/dG\s*rxn0|dG0\s*rxn/g, "dG0").replace(/dG\s*rxn/g, "dG")
       .replace(/dGo\b/g, "dG0").replace(/\bEo\b/g, "E0")
       .replace(/\s+/g, " ").trim();
  return s;
}

// Reads a run of units starting at i: "J/(K*mol)", "M^(-1) s^(-1)", "kJ/mol".
// After a "/" everything that follows in the run is in the denominator, as
// chemists write J/mol·K. Returns { raw, si, factor, end } or null.
function eqReadUnits(s, i, type) {
  var raw = {}, j = i, sign = 1, any = false, depth = 0, lastEnd = i;
  function skip(k) { while (s[k] === " ") k++; return k; }
  function term(k) {
    k = skip(k);
    for (var w = 0; w < EQ_UNIT_WORDS.length; w++) {
      var word = EQ_UNIT_WORDS[w];
      if (s.substr(k, word.length) !== word) continue;
      var after = s[k + word.length];
      if (after && /[A-Za-z0-9]/.test(after) && !(after === "-" )) {
        if (!/^[-]1(?![0-9.])/.test(s.slice(k + word.length))) continue;
      }
      var e = 1, m;
      var rest = s.slice(k + word.length);
      if ((m = /^\^(?:\(\s*([-+]?\d+)\s*\)|([-+]?\d+))/.exec(rest))) { e = parseInt(m[1] || m[2], 10); rest = rest.slice(m[0].length); k += m[0].length; }
      else if (/^(M|s|mol|min)$/.test(word) && (m = /^-1(?![0-9.])/.exec(rest))) { e = -1; k += 2; }
      var unit = word;
      if (unit === "C" && type !== "nernst") unit = "degC";   // "25 C" in a thermal problem
      return { unit: unit, e: e, end: k + word.length };
    }
    return null;
  }
  var t = term(j);
  if (!t) return null;
  while (t) {
    raw[t.unit] = (raw[t.unit] || 0) + sign * t.e;
    any = true; j = t.end; lastEnd = j;
    var k = skip(j), c = s[k];
    if (c === ")" && depth > 0) { depth--; j = k + 1; lastEnd = j; if (depth === 0) sign = sign; k = skip(j); c = s[k]; }
    if (c === "/" || c === "*") {
      var k2 = skip(k + 1), open = false;
      if (s[k2] === "(") { open = true; k2 = skip(k2 + 1); }
      var nt = term(k2);
      if (!nt) break;
      if (c === "/") sign = -1;
      if (open) depth++;
      t = nt;
      continue;
    }
    var nj = term(k);
    if (nj) { t = nj; continue; }              // juxtaposed: "J mol^(-1)", "M^(-1)s^(-1)"
    break;
  }
  while (depth > 0) return null;              // an unclosed unit bracket: leave it to the bracket check
  var si = {}, factor = 1;
  for (var u in raw) {
    si = eqVecAdd(si, eqVecScale(EQ_UNIT_SI[u], raw[u]), 1);
    if (EQ_UNIT_FACTOR[u]) factor *= Math.pow(EQ_UNIT_FACTOR[u], raw[u]);
  }
  return any ? { raw: eqVecAdd(raw, {}, 1), si: si, factor: factor, end: lastEnd } : null;
}

// A bare number that is plainly a measured quantity (not a count like 2, not a
// power of ten) should have carried units.
function eqLooksPhysical(v) {
  if (Number.isInteger(v) && Math.abs(v) <= 10) return false;
  var l = Math.log10(Math.abs(v));
  if (Math.abs(l - Math.round(l)) < 1e-9) return false;
  return true;
}

function eqTokenize(s, spec) {
  var names = spec.names.concat(Object.keys(EQ_FUNCS)).sort(function (a, b) { return b.length - a.length; });
  var out = [], i = 0;
  while (i < s.length) {
    var c = s[i];
    if (c === " ") { i++; continue; }
    var num = /^(\d+\.?\d*|\.\d+)(?:[eE][-+]?\d+)?/.exec(s.slice(i));
    if (num) {
      var tok = { t: "num", v: parseFloat(num[0]) };
      i += num[0].length;
      var u = eqReadUnits(s, i, spec.type);
      if (u) { tok.raw = u.raw; tok.si = u.si; tok.factor = u.factor; i = u.end; }
      out.push(tok);
      continue;
    }
    if ("+-*/^()".indexOf(c) > -1) { out.push({ t: c }); i++; continue; }
    var hit = null;
    for (var n = 0; n < names.length; n++) if (s.substr(i, names[n].length) === names[n]) { hit = names[n]; break; }
    if (hit) { out.push(EQ_FUNCS[hit] ? { t: "fn", v: hit } : { t: "id", v: hit }); i += hit.length; continue; }
    if (c === "x" || c === "X") { out.push({ t: "*" }); i++; continue; }   // "x" used as a times sign
    return { error: "symbol", at: s.slice(i, i + 6) };
  }
  return { tokens: out };
}

// Recursive descent to a small tree. Written chemistry reads "dG0/RT" as
// dG0/(R T), so a product written by juxtaposition binds tighter than an
// explicit * or /; "dG0/R*T" with the star is still (dG0/R)*T.
function eqParse(tokens) {
  var p = 0;
  function peek() { return tokens[p]; }
  function startsFactor(tk) { return tk && (tk.t === "num" || tk.t === "id" || tk.t === "fn" || tk.t === "("); }
  function expr() {
    var left = term();
    while (peek() && (peek().t === "+" || peek().t === "-")) { var op = tokens[p++].t; left = { k: "bin", op: op, l: left, r: term() }; }
    return left;
  }
  function term() {
    var left = juxt();
    while (peek() && (peek().t === "*" || peek().t === "/")) { var op = tokens[p++].t; left = { k: "bin", op: op, l: left, r: juxt() }; }
    return left;
  }
  function juxt() {
    var left = unary();
    while (startsFactor(peek())) left = { k: "bin", op: "*", l: left, r: power() };
    return left;
  }
  function unary() {
    if (peek() && peek().t === "-") { p++; return { k: "neg", a: unary() }; }
    if (peek() && peek().t === "+") { p++; return unary(); }
    return power();
  }
  function power() {
    var base = primary();
    if (peek() && peek().t === "^") { p++; return { k: "pow", b: base, e: unary() }; }
    return base;
  }
  function primary() {
    var tk = tokens[p];
    if (!tk) throw "end";
    if (tk.t === "num") { p++; return { k: "num", tok: tk }; }
    if (tk.t === "id") { p++; return { k: "id", name: tk.v }; }
    if (tk.t === "fn") {
      p++;
      var arg = (peek() && peek().t === "(") ? primary() : power();
      return { k: "fn", f: tk.v, a: arg };
    }
    if (tk.t === "(") {
      p++;
      var inner = expr();
      if (!peek() || peek().t !== ")") throw "paren";
      p++;
      return inner;
    }
    throw "token";
  }
  var tree = expr();
  if (p !== tokens.length) throw (tokens[p].t === ")" ? "paren" : "token");
  return tree;
}

// Value of a tree. `mode` "raw" uses numbers as typed; "si" converts numbers
// that carry units; "c2k" adds 273.15 to numbers written in °C.
function eqValue(n, env, tr, mode) {
  switch (n.k) {
    case "num":
      var v = n.tok.v;
      if (mode === "si" && n.tok.factor) v *= n.tok.factor;
      if (mode === "c2k" && n.tok.raw && n.tok.raw.degC === 1) v += 273.15;
      return v;
    case "id": return n.name === "e" ? Math.E : env[n.name];
    case "neg":
      if (mode === "c2k" && n.a.k === "num" && n.a.tok.raw && n.a.tok.raw.degC === 1) return -n.a.tok.v + 273.15;
      return -eqValue(n.a, env, tr, mode);
    case "fn": return EQ_FUNCS[n.f](eqValue(n.a, env, tr, mode));
    case "pow": return Math.pow(eqValue(n.b, env, tr, mode), eqValue(n.e, env, tr, mode));
    case "bin":
      var a = eqValue(n.l, env, tr, mode), b = eqValue(n.r, env, tr, mode);
      if (n.op === "+") { tr(a); tr(b); return a + b; }
      if (n.op === "-") { tr(a); tr(b); return a - b; }
      return n.op === "*" ? a * b : a / b;
  }
}

// Units of a tree: { raw, si, sym } or null when unknown. `sym` marks units that
// come from a named symbol, which has no raw spelling, so only SI is compared.
function eqDim(n, notes) {
  function cmp(a, b, where) {
    if (!a || !b) return;
    if (a.sym || b.sym) { if (!eqVecSame(a.si, b.si)) notes.push({ kind: "DONT_CANCEL", where: where }); return; }
    if (eqVecSame(a.raw, b.raw)) return;
    notes.push({ kind: eqVecSame(a.si, b.si) ? "NOT_CONVERTED" : "DONT_CANCEL", where: where });
  }
  function dimensionless(d, where, allowConc) {
    if (!d) return;
    if (d.sym ? eqVecEmpty(d.si) : eqVecEmpty(d.raw)) return;
    // ln[A] with [A] in M (or a pressure) is how the textbook writes it
    var keys = Object.keys(d.raw);
    if (allowConc && keys.length === 1 && d.raw[keys[0]] === 1 && /^(M|atm|mmHg|torr)$/.test(keys[0])) return;
    notes.push({ kind: (!d.sym && eqVecEmpty(d.si)) ? "NOT_CONVERTED" : "DONT_CANCEL", where: where });
  }
  switch (n.k) {
    case "num":
      if (n.tok.raw) return { raw: n.tok.raw, si: n.tok.si, sym: false };
      if (eqLooksPhysical(n.tok.v)) { notes.bare = (notes.bare || 0) + 1; return null; }
      return { raw: {}, si: {}, sym: false };
    case "id":
      var si = EQ_SYMBOL_SI[n.name];
      return si === undefined || si === null ? null : { raw: si, si: si, sym: true };
    case "neg": return eqDim(n.a, notes);
    case "fn":
      if (n.a.k === "num" && !n.a.tok.raw) return { raw: {}, si: {}, sym: false };   // log(0.025): a ratio already taken
      dimensionless(eqDim(n.a, notes), "inside " + n.f + "( )", n.f === "ln" || n.f === "log");
      return { raw: {}, si: {}, sym: false };
    case "pow":
      var b = eqDim(n.b, notes), e = eqDim(n.e, notes);
      dimensionless(e, "in an exponent", false);
      if (!b) return null;
      var ex = (n.e.k === "num") ? n.e.tok.v : (n.e.k === "neg" && n.e.a.k === "num") ? -n.e.a.tok.v : null;
      if (ex === null) return (eqVecEmpty(b.si)) ? b : null;
      return { raw: eqVecScale(b.raw, ex), si: eqVecScale(b.si, ex), sym: b.sym };
    case "bin":
      var l = eqDim(n.l, notes), r = eqDim(n.r, notes);
      if (n.op === "+" || n.op === "-") { cmp(l, r, "between two terms that are added or subtracted"); return l || r; }
      if (!l || !r) return null;
      var sgn = n.op === "*" ? 1 : -1;
      return { raw: eqVecAdd(l.raw, r.raw, sgn), si: eqVecAdd(l.si, r.si, sgn), sym: l.sym || r.sym };
  }
  return null;
}

// Where the brackets fail to pair, in the student's own text.
function eqBracketProblem(text) {
  var stack = [];
  for (var i = 0; i < text.length; i++) {
    if (text[i] === "(") stack.push(i);
    else if (text[i] === ")") {
      if (!stack.length) return { kind: "extra_close", near: text.slice(Math.max(0, i - 14), i + 1).trim() };
      stack.pop();
    }
  }
  if (stack.length) return { kind: "unclosed", near: text.slice(stack[0], stack[0] + 18).trim() };
  return null;
}

// The equation-bearing parts of a message. A part with ordinary words in it is
// prose, not an equation, and is skipped rather than reported as unreadable.
var EQ_SHORT_WORDS = /^(?:is|it|be|so|if|oh|an|as|at|by|in|my|no|or|to|up|we|me|do|of|on|am|go|he|us|ok)$/i;
var EQ_SYMBOL_WORDS = /^(?:ln|log|exp|sqrt|Ea|Ecell|RT|nF|kJ|mol|mmHg|torr|atm|min|degC|Hvap|dHvap|dG|dH|Ka|Kb|Kc|Kp|Ksp|Kw|pH|pOH|pKa|At|A0|E0)$/;

// The equations in a message. Students write them inside sentences ("It would be
// ln(0.175/0.050) = -(Ea/8.314)(1/298 - 1/328). Is that right?"), so each
// sentence is cut down to the longest run of pieces that are not English words
// and contains "=". Sentence punctuation at the end is dropped.
function eqCandidates(message) {
  var chunks = String(message || "").split(/[\n;]|(?:[.?!]\s+)(?=[A-Z])/);
  var out = [];
  chunks.forEach(function (c) {
    if (c.indexOf("=") < 0) return;
    var pieces = c.trim().split(/\s+/);
    // Prose: a short common word (is, it, so, be...), or a word of three or more
    // letters that is all lowercase or Capitalised. Symbols such as kt, RT, nF,
    // Ea, dG0 or RTlnQ are not.
    function prose(w) {
      var b = w.replace(/^[("']+|[.,?!:;'")]+$/g, "");
      if (!/^[A-Za-z']{2,}$/.test(b) || EQ_SYMBOL_WORDS.test(b)) return false;
      if (b.length === 2) return EQ_SHORT_WORDS.test(b);
      return /^[A-Za-z][a-z']+$/.test(b);
    }
    var best = null, run = [];
    function close() {
      if (run.length && run.join(" ").indexOf("=") > -1) {
        var txt = run.join(" ");
        if (!best || txt.length > best.length) best = txt;
      }
      run = [];
    }
    pieces.forEach(function (w) { if (prose(w)) close(); else run.push(w); });
    close();
    if (best) out.push(best.replace(/^[,:;]+|[.,?!:;]+$/g, "").trim());
  });
  return out;
}

// { verdict, why?, near?, units? } or null when the message holds no equation
// about this problem's unknown. verdict: VALID, EQ_SIGN_FLIPPED,
// EQ_RECIPROCAL_MISSING, EQ_LOG_BASE, EQ_UNITS_NOT_CONVERTED, EQ_CELSIUS,
// EQ_WRONG, UNREADABLE.
function checkEquation(message, prob) {
  var spec = eqSpec(prob);
  if (!spec) return null;
  var parts = eqCandidates(message);
  var result = null;
  for (var c = 0; c < parts.length; c++) {
    var norm = eqNormalize(parts[c], spec.type);
    var sides = norm.split("=").map(function (x) { return x.trim(); }).filter(function (x) { return x.length; });
    if (sides.length < 2) continue;
    var whole = eqTokenize(norm.replace(/=/g, " "), spec);
    if (!whole.tokens || !whole.tokens.some(function (x) { return x.t === "id" && x.v === spec.unknown; })) {
      if (whole.error && new RegExp(spec.unknown).test(norm)) {
        result = { verdict: "UNREADABLE", why: "symbol", format: spec.format };
      }
      continue;   // the unknown must appear, or "T1 = 263 K" would pass as a setup
    }
    var br = eqBracketProblem(parts[c]);
    if (br) { result = { verdict: "UNREADABLE", why: "paren", near: br.near, bracket: br.kind, format: spec.format }; continue; }
    var trees = [];
    for (var s = 0; s < sides.length; s++) {
      var tks = eqTokenize(sides[s], spec);
      if (tks.error) { result = { verdict: "UNREADABLE", why: "symbol", format: spec.format }; trees = null; break; }
      if (tks.tokens.length === 1 && tks.tokens[0].t === "num") { trees.push(null); continue; }  // a bare final number
      try { trees.push(eqParse(tks.tokens)); }
      catch (err) { result = { verdict: "UNREADABLE", why: "form", format: spec.format }; trees = null; break; }
    }
    if (!trees) continue;
    for (var k = 0; k + 1 < trees.length; k++) {
      if (!trees[k] || !trees[k + 1]) continue;
      var v = eqJudge(trees[k], trees[k + 1], spec);
      var notes = [];
      var dl = eqDim(trees[k], notes), dr = eqDim(trees[k + 1], notes);
      if (dl && dr) {
        if (dl.sym || dr.sym) { if (!eqVecSame(dl.si, dr.si)) notes.push({ kind: "DONT_CANCEL", where: "between the two sides of the equation" }); }
        else if (!eqVecSame(dl.raw, dr.raw)) notes.push({ kind: eqVecSame(dl.si, dr.si) ? "NOT_CONVERTED" : "DONT_CANCEL", where: "between the two sides of the equation" });
      }
      var withUnits = (norm.match(/\d\s*(kJ|J|mol|K|s|M|L|V|atm|mmHg|torr|min|degC|C)\b/g) || []).length;
      // An equation copied from the Equations list carries constants such as 0.0592
      // with no units; nudging about units there is noise. Only once the student has
      // substituted a value of their own does a bare number earn the nudge.
      var substituted = extractNumbers(parts[c]).some(function (x) {
        return eqLooksPhysical(x) && [0.0592, 8.314, 96485, 0.08206, 273.15, 2.303, 0.693].every(function (k) { return Math.abs(x - k) > 1e-9; });
      });
      if (!substituted) notes.bare = 0;
      var firstOfKind = notes.filter(function (x, i) { return notes.findIndex(function (y) { return y.kind === x.kind; }) === i; });
      v.units = { bare: notes.bare || 0, carried: withUnits, problems: firstOfKind,
                  celsius: /degC/.test(norm) };
      return v;
    }
  }
  return result;
}

function eqHolds(left, right, env, mode) {
  var maxAbs = 0;
  function track(x) { if (isFinite(x) && Math.abs(x) > maxAbs) maxAbs = Math.abs(x); return x; }
  var L = eqValue(left, env, track, mode), R = eqValue(right, env, track, mode);
  if (!isFinite(L) || !isFinite(R)) return false;
  var scale = Math.max(Math.abs(L), Math.abs(R));
  if (scale < 1e-9 * maxAbs) scale = maxAbs;
  // 1%: loose enough for three-figure numbers typed by hand (263 for 263.15 moves
  // 1/T1 - 1/T2 by 0.1%), tight enough to see a flipped term that is small beside
  // the rest (Clausius-Clapeyron with P2 close to P1 differs by under 3%).
  return Math.abs(L - R) <= 0.01 * scale + 1e-12;
}

function eqJudge(left, right, spec) {
  var envs = [spec.env];
  // Energies may be carried in kJ as well as J; accept a consistent choice of either.
  var kj = {};
  ["Ea", "dH", "dG", "dG0"].forEach(function (n) { if (n in spec.env) kj[n] = spec.env[n] / 1000; });
  if (Object.keys(kj).length) envs.push(Object.assign({}, spec.env, kj));
  function withUnknown(env, value) {
    var e = Object.assign({}, env);
    e[spec.unknown] = value / ((env !== spec.env && (spec.unknown in kj)) ? 1000 : 1);
    return e;
  }
  function any(mode, env) { for (var i = 0; i < envs.length; i++) if (eqHolds(left, right, env ? env(envs[i]) : envs[i], mode)) return true; return false; }
  if (any("raw")) return { verdict: "VALID" };
  for (var f = 0; f < spec.flips.length; f++) {
    var fv = spec.flips[f].value;
    if (any("raw", function (e) { return withUnknown(e, fv); })) return { verdict: spec.flips[f].id };
  }
  // Correct once the student's own units are converted: the setup is right and
  // the units were never made to agree (kJ beside J, minutes beside seconds).
  if (any("si")) return { verdict: "EQ_UNITS_NOT_CONVERTED" };
  if (any("c2k")) return { verdict: "EQ_CELSIUS" };
  return { verdict: "EQ_WRONG" };
}

// Answer-key values in a reply that the student has not produced, for the step
// being worked and every later one. Givens in the problem text and small whole
// numbers (orders, coefficients) are never counted.
function unearnedValues(reply, prob, currentMilestone, studentText) {
  var said = extractNumbers(studentText || ""), inReply = extractNumbers(reply || "");
  var givens = extractNumbers((prob && prob.text) || "");
  var out = [];
  for (var m = (currentMilestone || 0) + 1; m <= 5; m++) {
    var t = targetsFor(prob, m);
    if (!t) continue;
    for (var i = 0; i < t.values.length; i++) {
      var v = t.values[i];
      if (!isFinite(v) || (Math.abs(v) <= 10 && Math.abs(v - Math.round(v)) < 1e-9)) continue;
      var one = { values: [v], tol: t.tol, absTol: t.absTol, exact: t.exact,
                  absTols: t.absTols ? [t.absTols[i]] : undefined };
      var hit = function (x) { return numberHits(x, one); };
      if (inReply.some(hit) && !said.some(hit) && !givens.some(hit)) {
        out.push({ label: t.label, value: +v.toPrecision(3) });
      }
    }
  }
  return out;
}

// The setup handed over inside the tutor's own question, while the problem is
// being worked (steps 2 to 4): an algebraic change or ratio the student has not
// written ("decreases by x", "1:1"), or two or more givens the student has not
// yet used ("using 0.100 M and Kc = 0.025, what is…"). Simulated students: 5 of
// 120 replies. Returns a short description, or "".
var SETUP_RE = /(?:[+\-−]\s*\d*\s*[xs]\b|\bby\s+\d*[xs]\b|\b\d[xs]\b|\b[xs]²|\(\d*[xs]\)|\b\d+\s*:\s*\d+\b|\b\d-to-\d\b)/;
function handedSetup(reply, prob, currentMilestone, studentText) {
  if (currentMilestone < 1 || currentMilestone > 3) return "";
  var q = String(reply || "").split(/(?<=[.?!])\s+/).filter(function (x) { return x.indexOf("?") > -1; }).join(" ");
  if (!q) return "";
  var m = q.match(SETUP_RE);
  if (m && !SETUP_RE.test(studentText || "")) return "the setup expression \"" + m[0].trim() + "\"";
  var said = extractNumbers(studentText || "");
  var near = function (a, b) { return Math.abs(a - b) <= 0.001 * Math.abs(b); };
  // A given written with a decimal point is a measurement, however small: "-10.0 °C"
  // is one of the two temperatures the student is meant to convert, and handing both
  // of them over inside the question is the thing this guard exists to catch.
  var givenNums = extractNumbers((prob && prob.text) || "");
  var givens = givenNums.filter(function (v) { return Math.abs(v) > 10 || sigFigsOf(v) >= 2; });
  var inQ = extractNumbers(q);
  var fed = givens.filter(function (g) { return inQ.some(function (x) { return near(x, g); }) && !said.some(function (x) { return near(x, g); }); });
  return fed.length >= 2 ? "which values from the problem to use (" + fed.join(", ") + ")" : "";
}

// The independent reading. Returns { error, quote } or null. Fails open: any
// problem with the call means the tutor's judgment stands, as before.
function findStudentError(message, prob, step, reply) {
  var key = PropertiesService.getScriptProperties().getProperty("GEMINI_API_KEY");
  var stepText = stepLines(prob && prob.key)[step] || "";
  var sys = "You check one student message in a General Chemistry II tutoring session for statements about the chemistry " +
    "that are false. You are not the tutor and you do not grade the step. Ignore missing detail, rough wording, informal " +
    "language, and anything merely incomplete: flag only a claim a chemistry teacher would mark as wrong (a false cause, a " +
    "false mechanism, a misconception). Respond only with JSON: {\"false_claim\": true or false, \"quote\": \"the exact words " +
    "from the student's message that state it, copied character for character\", \"why\": \"one sentence\"}.";
  var user = "Step the student is working on: " + stepText + "\n\nStudent's message: " + message +
    "\n\n(For context only, the tutor drafted this reply: " + reply + ")";
  var payload = {
    system_instruction: { parts: [{ text: sys }] },
    contents: [{ role: "user", parts: [{ text: user }] }],
    generationConfig: { responseMimeType: "application/json", temperature: 0 }
  };
  var url = "https://generativelanguage.googleapis.com/v1beta/models/" + PRIMARY_MODEL + ":generateContent?key=" + key;
  var r = UrlFetchApp.fetch(url, { method: "post", contentType: "application/json", payload: JSON.stringify(payload), muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) return null;
  var text = JSON.parse(r.getContentText()).candidates[0].content.parts[0].text;
  var j = JSON.parse(String(text).replace(/```json|```/g, "").trim());
  return judgeFinding(j, message);
}

// Accept the reading only if the quoted words are really in the student's message.
function judgeFinding(j, message) {
  if (!j || j.false_claim !== true || !j.quote) return null;
  var norm = function (t) { return String(t).toLowerCase().replace(/[^a-z0-9]+/g, " ").trim(); };
  var q = norm(j.quote);
  if (q.length < 8 || norm(message).indexOf(q) < 0) return null;
  return { quote: String(j.quote).trim(), why: String(j.why || "").trim() };
}

function errorNote(found, draft) {
  return "\n\nREWRITE: the student's message contains a false statement: \"" + found.quote + "\"" +
    (found.why ? " (" + found.why + ")" : "") + ". Your draft let it stand or restated their account as correct. " +
    "Write the reply again. Do not affirm the step, do not restate their account in corrected form as if it were theirs, " +
    "and do not explain the right answer. Credit, in a few words, any part that is right, then ask ONE question that makes " +
    "them test the false statement against something they already know. Set milestones_demonstrated to 0. Your draft was: " +
    JSON.stringify(draft);
}

function leakNote(leaked, draft, handed) {
  var parts = [];
  if (leaked.length) parts.push("stated " + leaked.map(function (l) { return l.value + " (" + l.label + ")"; }).join(" and ") +
    ", which the student has NOT produced");
  if (handed) parts.push("handed over " + handed + " inside your question, which is the student's work to do");
  return "\n\nREWRITE: your draft reply " + parts.join("; and it ") + ". Write the reply again without giving that away: " +
    "ask for it instead. Your draft was: " + JSON.stringify(draft);
}

// Instructor, 16 Sept: "when it seems the student is floundering or excelling, as
// a bonus opportunity" — reflect on how the concepts fit together, or offer an
// analogy or metaphor for the process. No schedule. Step 5 is already
// metacognitive and step 1 is the account gate, so steps 2 to 4 only. The server
// supplies the signal (the help ladder going up, or no help needed so far); the
// tutor decides whether the moment is right.
function reflectionNote(sc, currentMilestone) {
  if (currentMilestone < 1 || currentMilestone > 3) return "";
  var signal = "";
  if (sc.rung >= 2) {
    signal = "The student is struggling here (help level " + sc.rung + "). If this reply finds them getting it, " +
      "or if pushing on the chemistry is not working, a step back may help more than another push.";
  } else if ((sc.worst || 0) === 0 && currentMilestone >= 2) {
    signal = "The student has needed no help on this problem so far.";
  } else {
    return "";
  }
  return "\n\nREFLECTION (a bonus opportunity, never a requirement): " + signal + " If it fits naturally, you may spend your " +
    "one question on reflection instead of the next chemistry question: how the ideas in this problem fit together (what the " +
    "particles are doing, what the symbols stand for, what an instrument would read), or an analogy or metaphor of their own " +
    "for the process. Skip it when the chemistry question matters more, and never ask for reflection in two replies running. " +
    "A reflection answer is not graded and clears nothing; an analogy is welcome but is never an account on its own.";
}

// Instructor, 16 Sept: the caret "is a menace. It has no mathematical priority
// and on a calculator can help you fail." When a student types one, steer them
// to the toolbar superscripts, gently, in passing.
function caretNote(message) {
  if (String(message || "").indexOf("^") < 0) return "";
  return "\n\nCARET: the student typed ^ for a power. In one short clause, gently steer them to the superscript buttons " +
    "in the toolbar (such as ² or ⁻¹): ^ has no agreed priority, and a calculator can read it differently from what they mean. " +
    "This is a habit, not an error: do not withhold anything for it, and do not make it your question unless nothing else needs asking.";
}

// Column 16 when the numeric grader named nothing: the equation's verdict, or,
// for a valid equation, what its units need.
function equationTrapId(eq) {
  if (!eq) return "";
  if (eq.verdict === "UNREADABLE") return eq.why === "paren" ? "EQ_BRACKETS" : "EQ_UNREADABLE";
  if (eq.verdict !== "VALID") return eq.verdict;
  var u = eq.units;
  if (u && u.problems.length) return "EQ_UNITS_" + u.problems[0].kind;
  if (u && u.bare > 0) return "EQ_UNITS_MISSING";
  return "";
}

// What the tutor is told. Correct what was submitted; do not project ahead.
function equationBlock(eq, prob) {
  if (!eq) return "";
  var head = "\n\nEQUATION CHECK (the server substituted this problem's own values, including the correct answer, into the equation the student typed):\n";
  if (eq.verdict === "UNREADABLE") {
    if (eq.why === "paren") {
      return head + (eq.bracket === "unclosed"
        ? "The brackets do not pair up: the opening bracket at \"" + eq.near + "\" is never closed. "
        : "The brackets do not pair up: there is a closing bracket with no opening one, near \"" + eq.near + "\". ") +
        "That is a real error, not a typo: a calculator reads unpaired brackets differently from what they meant. " +
        "Point to that bracket and ask what it is meant to enclose. Do not judge the rest of the equation yet and do not confirm it.";
    }
    return head + "The server could not read it. Ask the student to retype the equation on one line: * for times, / for divide, " +
      "the toolbar superscripts (such as ² or ⁻¹) for powers, ln( ) or log( ) with brackets, each number followed by its units " +
      "(the toolbar also has × 10), and these names: " + eq.format + ". " +
      "Do not judge the equation yourself and do not confirm it.";
  }
  var out = head;
  if (eq.verdict === "VALID") {
    out += "It holds: this is a valid form of the relationship. You may confirm the EQUATION, not any number. " +
      "Any valid rearrangement is acceptable; do not ask them to rewrite it in another form.";
  } else {
    var what = {
      EQ_SIGN_FLIPPED: "It does NOT hold. It holds only if the answer has the opposite sign (or, for a logarithm, the inverted ratio): somewhere in their rearrangement a sign or the order of a subtraction or ratio was flipped.",
      EQ_RECIPROCAL_MISSING: "It does NOT hold. It holds only for the reciprocal of the answer: they solved for 1/[A] and stopped, or dropped a reciprocal.",
      EQ_LOG_BASE: "It does NOT hold. It holds only if log were read as ln: the 0.0592 form needs base-10 log.",
      EQ_UNITS_NOT_CONVERTED: "It does NOT hold as written, but it WOULD hold once the units they wrote are made to agree: the setup is right and a unit was never converted (kJ beside J, minutes beside seconds, mmHg beside atm).",
      EQ_CELSIUS: "It does NOT hold as written, but it WOULD hold with the temperatures in kelvin: a Celsius temperature was used where the gas constant needs kelvin.",
      EQ_WRONG: "It does NOT hold, and it is not one of the common slips (flipped sign, missing reciprocal, unconverted units)."
    }[eq.verdict] || "It does NOT hold.";
    out += what + "\nThis is algebra, and correcting what the student submitted is part of your job. Work on THEIR equation, one step: " +
      "point to the single place in their own expression where it goes wrong, show that one step done correctly in plain text, " +
      "and ask them to carry it through and retype the equation. Do not write out the finished rearrangement, and do not show " +
      "any step beyond the one that went wrong. Do NOT clear the milestone.";
  }
  var u = eq.units;
  if (u) {
    if (u.problems.length) {
      out += "\nUNITS: they do not cancel " + u.problems[0].where +
        (u.problems[0].kind === "NOT_CONVERTED" ? ", until a conversion the student has not made" : "") +
        ". A number only quantifies a unit or a dimension and has no meaning of its own; units cancelling is what gives the logarithm and the answer their meaning. So this is worth one question: point to that place and ask what units the pieces there carry.";
    } else if (u.bare > 0) {
      out += "\nUNITS: " + (u.bare === 1 ? "one number" : u.bare + " numbers") + " in it " + (u.bare === 1 ? "carries" : "carry") +
        " no units. Encourage units gently, never as a penalty: once the rest is settled, ask what units one of those quantities carries and what happens to them in the equation. " +
        "A number only quantifies a unit or a dimension; without its units it has no meaning, and on the exam it earns nothing.";
    } else if (u.carried > 0) {
      out += "\nUNITS: every number carries units and they cancel correctly. Say so in a few words; carrying units through is exactly the habit to reinforce.";
    }
    if (u.celsius && eq.verdict !== "EQ_CELSIUS") out += " A temperature is written in °C; in this equation it must be in kelvin.";
  }
  return out;
}

var DEFAULT_TOL = 0.02;   // 2% everywhere
var TIGHT_TOL   = 0.005;  // 0.5% for Raoult (see notes)

var R_J   = 8.314;    // J/(mol K)
var R_ATM = 0.08206;  // L atm/(mol K)

// ---------- 1. READING NUMBERS OUT OF A STUDENT MESSAGE ----------

function normalizeNumberText(s) {
  if (!s) return "";
  var t = String(s);

  // A power written on a symbol is not a number the student computed: "x²",
  // "(2x)²", "s³", "x^2". Found by the simulated students, 17 Sept: "x² / (0.500 - x)"
  // read as the number 2, which matched a pH of 2.02 and granted two steps.
  t = t.replace(/([A-Za-z\)\]])\s*[\u2070\u00B9\u00B2\u00B3\u2074-\u2079]+/g, "$1 ")
       .replace(/([A-Za-z\)\]])\s*\^\s*(?:\(\s*[+-]?\d+\s*\)|[+-]?\d+)/g, "$1 ");

  // "15,200" is fifteen thousand two hundred, not 15 and 200
  t = t.replace(/(\d),(\d{3})(?![\d.])/g, "$1$2");

  // unicode superscripts -> plain characters
  var sup = { "\u2070":"0","\u00B9":"1","\u00B2":"2","\u00B3":"3","\u2074":"4",
              "\u2075":"5","\u2076":"6","\u2077":"7","\u2078":"8","\u2079":"9",
              "\u207A":"+","\u207B":"-" };
  t = t.replace(/[\u2070\u00B9\u00B2\u00B3\u2074-\u2079\u207A\u207B]/g, function(c){ return sup[c]; });

  // oxidation states written in names: "Fe(III)" -> "Fe(3)"
  var roman = { I:1, II:2, III:3, IV:4, V:5, VI:6, VII:7, VIII:8 };
  t = t.replace(/\((VIII|VII|VI|IV|V|III|II|I)\)/g, function (m, r) { return "(" + roman[r] + ")"; });

  // reaction orders written as words: "zero order", "second-order", "first in B"
  var ordinal = { zeroth:0, zero:0, first:1, second:2, third:3 };
  t = t.replace(/\b(zeroth|zero|first|second|third)(?=[\s-]+(order|in\b|with respect))/gi,
    function (m, w) { return String(ordinal[w.toLowerCase()]); });

  // trial and experiment labels are not answers: "trials 1 and 3" -> "trial"
  t = t.replace(/\b(trials?|experiments?)\s*#?\s*\d+(?![\d.])(\s*(,|and|&|vs\.?|to|with)\s*(trials?|experiments?)?\s*#?\s*\d+(?![\d.]))*/gi, "$1");

  // multiplication signs and unicode minus -> ascii
  t = t.replace(/[\u00D7\u2217\u22C5\u2022]/g, "x").replace(/[\u2212\u2013\u2014]/g, "-");

  // "3.4 x 10^-2", "3.4*10-2", "3.4 X 10 -2"  ->  "3.4e-2"
  t = t.replace(/(\d(?:\.\d+)?)\s*[x*]\s*10(?![\d.])\s*\^?\s*([+-]?\d+)/gi, "$1e$2");

  // Digits written onto a formula or a symbol are not numbers the student
  // computed: H2O, H3O+, PCl5, Cu2+, k2, T1, [A]0, t1/2. Simulated students, 17 Sept:
  // "HCOOH + H2O" read as 2, matched a pH of 2.02 and jumped a student from step 1
  // to step 4. Scientific notation (1.8e-5) is already folded above and is kept.
  t = t.replace(/t\s*1\s*\/\s*2/g, "thalf");
  // Counts of particles and multipliers are not answers: "8 protons, 8 neutrons",
  // "8(1.0073 amu)". Simulated run, 17 Sept: the 8 matched a binding energy of
  // 8.005 MeV per nucleon. "3 unpaired electrons" and "7 d electrons" are kept.
  t = t.replace(/(^|[^\w.])\d{1,3}(?=\s*\()/g, "$1")
       .replace(/(^|[^\w.])\d{1,3}\s+(protons?|neutrons?|electrons?|nucleons?|atoms?|molecules?)\b/gi, "$1$2");
  // Orbital labels are not numbers: "[Ar] 3d⁷", "4s electrons", "2p". Simulated
  // run, 17 Sept: the 3 of "3d⁷" matched 3 unpaired electrons and granted steps 3
  // and 4 to a student who had only counted d electrons.
  t = t.replace(/(^|[^\w.])[1-7]([spdf])(?=[\s\u2070-\u2079\u00B2\u00B3\u00B9^,;:)\]]|$)/g, "$1$2");
  // A charge ("Cu2+", "Fe3+") is kept: oxidation states and electron counts are answers.
  t = t.replace(/([A-Za-z\]\)])(\d+)(?![\d.+\-])/g, function (m, L, d, off, str) {
    if ((L === "e" || L === "E") && /\d/.test(str.charAt(off - 1))) return m;
    return L + " ";
  });

  // unit exponents are not answers: "s-1", "M^-2", "mol-1" -> drop the exponent.
  // Needs a minus sign right after a letter, so "Fe+3", "Cu2+", "H2O", "dH -44"
  // and scientific notation like "1.8e-5" are left alone.
  t = t.replace(/(\d?)([A-Za-z])\^?(-\d+)/g, function (m, d, L) {
    if (d && (L === "e" || L === "E")) return m;
    return d + L + " ";
  });

  // strip thousands separators: 64,600 -> 64600
  t = t.replace(/(\d),(?=\d{3}(\D|$))/g, "$1");

  return t;
}

// Significant figures of each number as typed, by value, from the latest
// extraction. "0.10" is two figures though the float is 0.1.
var TYPED_SIG_FIGS = {};

function extractNumbers(s) {
  var t = normalizeNumberText(s);
  var out = [];
  TYPED_SIG_FIGS = {};   // only this message's numbers; an earlier "0.100" must not make "0.10" three figures
  var re = /[+-]?(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?/g;
  var m;
  while ((m = re.exec(t)) !== null) {
    var v = parseFloat(m[0]);
    if (isFinite(v)) {
      out.push(v);
      var mant = m[0].replace(/^[+-]/, "").replace(/[eE].*$/, "");
      var digits = mant.replace(".", "").replace(/^0+/, "");
      if (mant.indexOf(".") < 0) digits = digits.replace(/0+$/, "");
      TYPED_SIG_FIGS[String(v)] = Math.max(TYPED_SIG_FIGS[String(v)] || 0, digits.length);
    }
  }
  return out;
}

// ---------- 2. COMPARING A NUMBER TO A TARGET ----------
// target = { values:[...], tol:0.02, exact:false, ignoreSign:false }

// Significant figures of a number as typed (from its shortest exact form).
// Trailing zeros of a whole number are not counted.
function sigFigsOf(x) {
  if (!isFinite(x) || x === 0) return 0;
  if (TYPED_SIG_FIGS[String(x)]) return TYPED_SIG_FIGS[String(x)];
  for (var p = 1; p <= 15; p++) if (parseFloat(Math.abs(x).toPrecision(p)) === Math.abs(x)) break;
  if (Math.abs(x) === Math.round(Math.abs(x))) {
    var d = String(Math.round(Math.abs(x))).replace(/0+$/, "");
    return Math.min(p, d.length);
  }
  return p;
}

// Instructor, 17 Sept: "accept 1.3 × 10⁻⁵, try to gently tutor to the correct sig
// figs, but accept with a warning." True when `got` is the answer rounded to the
// figures the student gave, and they gave at least two.
function roundsTo(got, want) {
  var sf = sigFigsOf(got);
  if (sf < 2 || want === 0) return false;
  var step = Math.pow(10, Math.floor(Math.log10(Math.abs(got))) - sf + 1);
  return Math.abs(got - want) <= 0.5 * step * (1 + 1e-9);
}

// The fewest significant figures among the measured values in the problem text
// (numbers written with a decimal point or in scientific notation).
function dataSigFigs(prob) {
  var text = normalizeNumberText((prob && prob.text) || "");
  var re = /(\d*\.\d+|\d+\.\d*)(?:e[+-]?\d+)?|\d+e[+-]?\d+/gi, m, best = null;
  while ((m = re.exec(text)) !== null) {
    var mant = m[0].replace(/e.*$/i, "").replace(".", "").replace(/^0+/, "");
    var n = mant.length;
    if (n > 0 && (best === null || n < best)) best = n;
  }
  return best;
}

function numberHits(got, target, rounding) {
  // A single-digit whole number ("8 protons", "H2O", "2x") does not match an answer
  // that is not itself a whole number: 8 is not a binding energy of 7.98 MeV per
  // nucleon, and 2 is not a pH of 2.02. Whole-number answers (orders, n, pH 7.00)
  // are unaffected.
  if (!target.exact && got === Math.round(got) && Math.abs(got) < 10 && sigFigsOf(got) <= 1) {
    var anyWhole = false;
    for (var w = 0; w < target.values.length; w++) {
      var tv = target.values[w];
      if (isFinite(tv) && Math.abs(tv - Math.round(tv)) <= 1e-3 * Math.max(1, Math.abs(tv))) anyWhole = true;
    }
    if (!anyWhole) return false;
  }
  var tol = (target.tol === undefined) ? DEFAULT_TOL : target.tol;
  // Some problems have a given that sits inside the answer's tolerance — the
  // starting temperature in Clausius-Clapeyron is 1.4% from the final one. A
  // target may name such values in `notThese`; a number closer to one of those
  // than to the real answer is wrong however tight the window says it is.
  if (target.notThese && !target.exact) {
    var best = Infinity, bestIsWrong = false;
    for (var w = 0; w < target.values.length; w++) {
      var d = Math.abs(got - target.values[w]);
      if (d < best) { best = d; bestIsWrong = false; }
    }
    for (var z = 0; z < target.notThese.length; z++) {
      var dz = Math.abs(got - target.notThese[z]);
      if (dz < best) { best = dz; bestIsWrong = true; }
    }
    if (bestIsWrong) return false;
  }
  for (var i = 0; i < target.values.length; i++) {
    var exp = target.values[i];
    var a = target.ignoreSign ? Math.abs(got) : got;
    var b = target.ignoreSign ? Math.abs(exp) : exp;
    // absTols lets one value (a pH) use an absolute window while another in the
    // same list (a concentration) keeps the ordinary percentage window.
    var at = (target.absTols && target.absTols[i] != null) ? target.absTols[i]
           : (target.absTols ? undefined : target.absTol);
    if (target.exact) {
      if (Math.round(a * 1000) === Math.round(b * 1000)) return true;
    } else if (at !== undefined) {
      if (Math.abs(a - b) <= at) return true;
    } else if (b === 0) {
      if (Math.abs(a) < 1e-9) return true;
    } else if (Math.abs(a - b) <= tol * Math.abs(b)) {
      return true;
    } else if (rounding && roundsTo(a, b)) {
      return true;
    }
  }
  return false;
}

// why was it wrong? used only to steer the tutor's next question
function diagnose(got, target) {
  var exp = target.values[0];
  if (!exp) return "WRONG";
  var tol = (target.tol === undefined) ? DEFAULT_TOL : target.tol;
  // a sign error only counts if the signs genuinely differ
  if (!target.ignoreSign && (got < 0) !== (exp < 0) &&
      Math.abs(Math.abs(got) - Math.abs(exp)) <= tol * Math.abs(exp)) {
    return "SIGN_ERROR";
  }
  if (Math.abs(Math.abs(got - exp) - 273.15) < 1.5) return "CELSIUS_KELVIN";
  var ratios = [1000, 0.001, 100, 0.01, 10, 0.1, 60, 1/60, 3600, 1/3600, 760, 1/760];
  for (var i = 0; i < ratios.length; i++) {
    if (got !== 0 && Math.abs((exp / got) - ratios[i]) < 0.05 * Math.abs(ratios[i])) {
      return "UNIT_OR_CONVERSION";
    }
  }
  return "WRONG";
}

// Q inverted has a signature the solver can compute, so wire it to the answer.
function trapValues(prob, milestone, target) {
  var out = [];
  var key = prob && prob.key;
  var v = prob && prob.vars && prob.vars.solve;
  if (!key || !v || !TRAPS[key]) return out;
  var list = TRAPS[key];
  for (var i = 0; i < list.length; i++) {
    var val = null;
    try { val = list[i].value(v); } catch (e) { val = null; }
    if (val === null && list[i].id === "Q_INVERTED" && target && target.values[0]) {
      val = 1 / target.values[0];
    }
    if (val !== null && isFinite(val)) out.push({ id: list[i].id, value: val, say: list[i].say });
  }
  return out;
}

// Which named misconception does this wrong answer correspond to?
function namedTrap(got, prob, milestone, target) {
  var key = prob && prob.key;
  var v = prob && prob.vars && prob.vars.solve;

  // A predicate trap fires on a property of the answer rather than on a value we
  // predicted. It is checked FIRST, because "that cannot happen" is a stronger
  // and simpler objection than "here is the slip you made", and it is the one the
  // student should have raised themselves.
  if (key && v && TRAPS[key]) {
    var list = TRAPS[key];
    for (var i = 0; i < list.length; i++) {
      if (!list[i].test) continue;
      var hit = false;
      try { hit = list[i].test(got, v); } catch (e) { hit = false; }
      if (hit) {
        // The impossibility is the first question. But if we also know exactly
        // which slip produced it, hand that over as the follow-up, so the tutor
        // has it ready once the student has seen for themselves that the number
        // cannot be right. x^2/a0-x always overshoots a0, so without this the
        // specific diagnosis would be masked every single time.
        var also = null, vals = trapValues(prob, milestone, target);
        for (var m = 0; m < vals.length; m++) {
          if (vals[m].value !== 0 && Math.abs(got - vals[m].value) <= 0.02 * Math.abs(vals[m].value)) {
            also = vals[m]; break;
          }
        }
        return { id: list[i].id + (also ? " then " + also.id : ""), value: null,
                 say: list[i].say + (also ? " ONLY AFTER they have seen that it is impossible, and only if they " +
                      "ask what went wrong: " + also.say : "") };
      }
    }
  }

  var cands = trapValues(prob, milestone, target);
  for (var j = 0; j < cands.length; j++) {
    var c = cands[j];
    if (c.value !== 0 && Math.abs(got - c.value) <= 0.02 * Math.abs(c.value)) return c;
  }
  return null;
}

function collectNumbers(obj, out) {
  if (obj === null || obj === undefined) return out;
  if (typeof obj === "number") { out.push(obj); return out; }
  if (Object.prototype.toString.call(obj) === "[object Array]") {
    for (var i = 0; i < obj.length; i++) collectNumbers(obj[i], out);
    return out;
  }
  if (typeof obj === "object") {
    for (var k in obj) if (Object.prototype.hasOwnProperty.call(obj, k)) {
      collectNumbers(obj[k], out);
    }
    return out;
  }
  return out;
}

function knownValues(prob) {
  var vals = CHEM_CONSTANTS.slice();
  for (var n = 0; n <= 12; n++) vals.push(n);          // coefficients, counts
  // Anything printed in the problem itself counts too: trial data, volumes
  // written in mL, constants quoted in the text.
  if (prob && prob.text) {
    var shown = extractNumbers(prob.text);
    for (var q = 0; q < shown.length; q++) vals.push(shown[q]);
  }
  if (prob && prob.vars && prob.vars.solve) {
    var inputs = collectNumbers(prob.vars.solve, []);
    for (var i = 0; i < inputs.length; i++) vals.push(inputs[i]);
    // Students show their working. Ratios and products of the givens are the
    // ordinary intermediates of almost every chemistry calculation — moles from
    // mass over molar mass, molality, concentration ratios. Treat them as
    // accounted for so honest work is never questioned.
    for (var a = 0; a < inputs.length; a++) {
      for (var b = 0; b < inputs.length; b++) {
        if (a === b) continue;
        if (inputs[b] !== 0) vals.push(inputs[a] / inputs[b]);
        if (b > a) vals.push(inputs[a] * inputs[b]);
        if (b > a) vals.push(inputs[a] + inputs[b], Math.abs(inputs[a] - inputs[b]));  // total volumes, differences
      }
    }
    for (var m = 1; m <= 5; m++) {                      // every computed value
      var t = targetsFor(prob, m);
      if (t) for (var j = 0; j < t.values.length; j++) vals.push(t.values[j]);
      if (t && t.companions) for (var c = 0; c < t.companions.length; c++) vals.push(t.companions[c]);
    }
  }
  return vals;
}

function isAccountedFor(x, vals) {
  for (var i = 0; i < vals.length; i++) {
    var v = vals[i];
    if (v === 0) { if (Math.abs(x) < 1e-9) return true; continue; }
    var ax = Math.abs(x), av = Math.abs(v);
    if (Math.abs(ax - av) <= 0.03 * av) return true;
    // the same value with a unit prefix: mL vs L, mmol vs mol, J vs kJ
    if (Math.abs(ax - av * 1000) <= 0.03 * av * 1000) return true;
    if (Math.abs(ax - av / 1000) <= 0.03 * av / 1000) return true;
  }
  return false;
}

// Returns the first number in the message that matches nothing the student
// could justify, or null. Only called when the answer itself was correct — a
// student who is already wrong does not need a second question piled on.
function strayNumber(message, prob, accepted) {
  var vals = knownValues(prob);
  var nums = extractNumbers(message);
  for (var i = 0; i < nums.length; i++) {
    if (accepted !== undefined && nums[i] === accepted) continue;   // the answer itself, accepted by rounding
    if (!isAccountedFor(nums[i], vals)) return nums[i];
  }
  return null;
}

// Targets marked milli also accept the value in millimoles, but only when the
// student actually wrote "mmol". "2.50 mmol" is right; "2.50 mol" is not.
function withMilli(message, target) {
  if (!target || !target.milli || !/mmol/i.test(String(message))) return target;
  var t = {};
  for (var k in target) t[k] = target[k];
  t.values = target.values.concat(target.values.map(function (x) { return x * 1000; }));
  return t;
}

// A target marked `all` names several different quantities that the step needs
// together (two temperatures, two potentials). V1 accepted any ONE of them, and
// that is how "23K 293K" was graded correct: 293 matched and 23 was never read.
// Returns the numbers that matched, one per value, never reusing a number.
function matchAll(nums, target, used) {
  var picked = [];
  for (var i = 0; i < target.values.length; i++) {
    var one = {};
    for (var k in target) one[k] = target[k];
    one.values = [target.values[i]];
    one.notThese = undefined;
    if (target.absTols) one.absTols = [target.absTols[i]];
    var found = null;
    for (var j = 0; j < nums.length; j++) {
      if (picked.indexOf(j) > -1) continue;
      if (used && spentAlready(nums[j], used)) continue;
      if (numberHits(nums[j], one, true)) { found = j; break; }
    }
    if (found !== null) picked.push(found);
  }
  return picked.map(function (j) { return nums[j]; });
}

function gradeMessage(message, target, prob) {
  if (!target) return { verdict: "NO_KEY" };
  target = withMilli(message, target);

  var nums = extractNumbers(message);
  if (nums.length === 0) return { verdict: "NO_NUMBER", label: target.label };

  if (target.all) {
    var hit = matchAll(nums, target);
    if (hit.length === target.values.length) {
      var strayAll = prob ? strayNumber(message, prob) : null;
      if (strayAll !== null) {
        return { verdict: "CORRECT_WITH_STRAY", label: target.label, student: hit.join(" and "), stray: strayAll };
      }
      return { verdict: "CORRECT", label: target.label, student: hit.join(" and ") };
    }
    if (hit.length > 0) {
      return { verdict: "PARTIAL", label: target.label, student: hit.join(", "),
               needs: target.values.length, all: true };
    }
    // nothing matched: fall through to the ordinary wrong-answer diagnosis
  }

  var last = nums[nums.length - 1];
  if (numberHits(last, target, true)) {
    var rounded = numberHits(last, target) ? null : { sf: sigFigsOf(last), dataSf: dataSigFigs(prob) };
    var stray = prob ? strayNumber(message, prob, rounded ? last : undefined) : null;
    if (stray !== null) {
      return { verdict: "CORRECT_WITH_STRAY", label: target.label,
               student: last, stray: stray, rounded: rounded };
    }
    return { verdict: "CORRECT", label: target.label, student: last, rounded: rounded };
  }

  // Formula subscripts and charges written after the answer ("4.50e-3 mol NH4+",
  // "0.164 M Cl2") are not a second answer. For answers that are not whole-number
  // counts, look past trailing small whole numbers to find the real final value.
  {
    var j = nums.length - 1;
    while (j > 0 && (isCompanion(nums[j], target) || isGivenNotAnswer(nums[j], prob) ||
           (!target.exact && nums[j] === Math.round(nums[j]) && Math.abs(nums[j]) <= 12))) j--;
    if (j < nums.length - 1 && numberHits(nums[j], target, true)) {
      var s2 = prob ? strayNumber(message, prob) : null;
      if (s2 !== null) {
        return { verdict: "CORRECT_WITH_STRAY", label: target.label, student: nums[j], stray: s2 };
      }
      return { verdict: "CORRECT", label: target.label, student: nums[j] };
    }
  }

  // last one is wrong. If an earlier number matches the target, decide between
  // two readings: either both numbers are legitimate and we cannot tell which
  // is the final answer, or the last one came from nowhere and that is the more
  // useful thing to ask about.
  // "pH = -2.79": the magnitude is right and the sign is not. That is a slip to name,
  // not an ambiguity to ask about (pre-launch run, 17 Sept).
  if (numberHits(-last, target, true)) {
    var signTrap = namedTrap(last, prob, undefined, target);
    return { verdict: "WRONG", label: target.label, student: last, trap: signTrap, reason: signTrap ? signTrap.id : "SIGN_ERROR" };
  }

  for (var i = 0; i < nums.length - 1; i++) {
    if (numberHits(nums[i], target)) {
      if (prob && !isAccountedFor(last, knownValues(prob))) {
        return { verdict: "CORRECT_WITH_STRAY", label: target.label,
                 student: nums[i], stray: last };
      }
      return { verdict: "AMBIGUOUS", label: target.label, student: last };
    }
  }

  // No answer, but the message does hold a value that belongs alongside it —
  // the intermediate the answer is built from, or the constant it is compared
  // against. That is progress, not a wrong answer.
  for (var p = 0; p < nums.length; p++) {
    if (isCompanion(nums[p], target)) {
      return { verdict: "PARTIAL", label: target.label, student: nums[p] };
    }
  }

  // A trailing small whole number beside other numbers is a coefficient or a count
  // ("0.75 = 2x"), not the student's answer: no trap is named on it.
  var trivialLast = !target.exact && nums.length > 1 && last === Math.round(last) && Math.abs(last) <= 12;
  var named = trivialLast ? null : namedTrap(last, prob, undefined, target);
  return {
    verdict: "WRONG",
    label: target.label,
    student: last,
    trap: named,
    reason: named ? named.id : diagnose(last, target)
  };
}

// A value the student was handed — printed in the problem, one of the solve
// inputs, or a standard constant — that is not itself the answer to any
// milestone. Listing givens after an answer ("T1 = 351.55 K, dH = 38.6 kJ/mol")
// does not make the answer ambiguous. Values that ARE answers somewhere (the
// pure vapor pressure in Raoult, a rate constant) are excluded, so a student
// hedging between the right answer and a trap still gets asked.
function isGivenNotAnswer(x, prob) {
  if (!prob) return false;
  function near(a, b) { return b !== 0 && Math.abs(Math.abs(a) - Math.abs(b)) <= 0.01 * Math.abs(b); }
  for (var m = 1; m <= 5; m++) {
    var t = targetsFor(prob, m);
    if (t) for (var i = 0; i < t.values.length; i++) if (near(x, t.values[i])) return false;
  }
  var given = CHEM_CONSTANTS.slice();
  if (prob.vars && prob.vars.solve) given = given.concat(collectNumbers(prob.vars.solve, []));
  if (prob.text) given = given.concat(extractNumbers(prob.text));
  for (var k = 0; k < given.length; k++) if (near(x, given[k])) return true;
  return false;
}

// Companions are values that legitimately sit next to the answer (an
// intermediate, or the constant it is compared against). They never count as
// the answer on their own.
function isCompanion(x, target) {
  if (!target || !target.companions) return false;
  return numberHits(x, { values: target.companions, ignoreSign: true });
}

// ---------- 4. THE SOLVERS ----------
// Each returns an object: milestone number -> target spec.

function targetsFor(prob, milestone) {
  if (!prob || !prob.vars || !prob.vars.solve) return null;
  var inp = prob.vars.solve;
  var fn = SOLVERS[inp.type];
  if (!fn) return null;
  var map;
  try { map = fn(inp); } catch (e) { return null; }

  // Milestone 1 is conceptual in every archetype. The problem text is not on the
  // student's screen until milestone 1 clears (see rule 3y), so a value read off
  // the problem cannot be asked for there. Four archetypes used to do exactly
  // that — reaction order, ion charge, d-electron count, mass defect. The value
  // moves to milestone 2, where the problem is visible: outright if milestone 2
  // has no target of its own, otherwise as a partial-credit companion of it.
  if (map[1]) {
    if (!map[2]) { map[2] = map[1]; }
    else {
      var comp = (map[2].companions || []).concat(map[1].values);
      map[2] = T(map[2].label, map[2].values, mergeOpts(map[2], comp));
    }
    delete map[1];
  }

  var t = map[milestone];
  if (!t) return null;
  // If a solve block is malformed the numbers come out NaN. Stand down rather
  // than marking a student wrong against a value that does not exist.
  for (var i = 0; i < t.values.length; i++) {
    if (typeof t.values[i] !== "number" || !isFinite(t.values[i])) return null;
  }
  if (t.absTol !== undefined && !isFinite(t.absTol)) return null;
  if (t.companions) for (var c = 0; c < t.companions.length; c++) {
    if (typeof t.companions[c] !== "number" || !isFinite(t.companions[c])) return null;
  }
  return t;
}

function mergeOpts(t, companions) {
  var o = {};
  for (var k in t) if (k !== "label" && k !== "values") o[k] = t[k];
  o.companions = companions;
  return o;
}

function T(label, values, opts) {
  var t = { label: label, values: [].concat(values) };
  if (opts) for (var k in opts) t[k] = opts[k];
  return t;
}

// An ICE problem names the species it wants. The solver used to offer every value
// it had computed as acceptable, so a student who solved correctly and then read
// off the wrong species passed: 0.061 M is the equilibrium [PCl5] and the problem
// asked for [PCl3] (0.039 M), and milestone 4 cleared. Where the problem says
// which one it wants, `ask` names it — and every other value becomes a value the
// target must REJECT, so reading the wrong species is caught rather than allowed.
function icePick(v, values, label) {
  var want = v.ask;
  if (!want || values[want] === undefined) {
    var all = [];
    for (var k in values) all.push(values[k]);
    return T(label, all);
  }
  var reject = [];
  for (var k2 in values) if (k2 !== want && values[k2] !== values[want]) reject.push(values[k2]);
  return T(label + " (" + want + ")", [values[want]], { notThese: reject });
}

var SOLVERS = {

  // Clausius-Clapeyron: ln(P2/P1) = -dH/R (1/T2 - 1/T1)
  "cc": function (v) {
    var T1 = v.tC + 273.15;
    var invT2 = 1 / T1 - (R_J / (v.dh * 1000)) * Math.log(v.p2 / v.p1);
    var T2 = 1 / invT2;
    return {
      2: T("starting temperature in Kelvin", [T1, v.tC]),
      // T1 can sit within 2% of T2. Without notThese, restating the starting
      // temperature would be graded as the answer.
      4: T("final temperature", [T2, T2 - 273.15], { notThese: [T1, v.tC] })
    };
  },

  // Henry: C = kH * P
  "henry": function (v) {
    return { 4: T("dissolved concentration", [v.kh * v.p]) };
  },

  // Raoult: P = X_solvent * P_pure   (tight tolerance, see notes)
  "raoult": function (v) {
    var nS = v.mSol / v.mmSol, nW = v.mSolv / v.mmSolv;
    var X = nW / (nW + nS);
    return {
      2: T("mole fraction of solvent", [X, 1 - X], { tol: TIGHT_TOL }),
      3: T("vapor pressure of pure solvent", [v.pPure], { tol: TIGHT_TOL }),
      4: T("vapor pressure of the solution", [X * v.pPure], { tol: TIGHT_TOL })
    };
  },

  // Colligative: dT = i * K * m   (grade the CHANGE, not the final reading)
  "colligative": function (v) {
    var m = (v.mSol / v.mmSol) / v.kgSolv;
    var dT = v.i * v.K * m;
    var final = v.base + v.dir * dT;
    return {
      2: T("van't Hoff factor", [v.i], { exact: true }),
      3: T("molality", [m]),
      // milestone 4: calculate dT, then apply it. The final reading is graded on a
      // window scaled to dT (never tighter than 0.06 so 101.05 -> 101.1 passes).
      4: T("the new freezing or boiling point", [final],
           { absTol: Math.max(0.02 * dT, 0.06), companions: [dT] })
    };
  },

  // Osmotic: pi = iMRT, then molar mass
  "osmotic": function (v) {
    var Tk = v.tC + 273.15;
    var atm = v.mmHg / 760;
    var M = atm / ((v.i || 1) * R_ATM * Tk);
    var mol = M * v.V;
    return {
      // Both conversions are the step. V1 labelled this "pressure in atm" and
      // accepted the temperature alone, so "298 K" cleared a pressure step.
      2: T("pressure in atm and temperature in Kelvin", [atm, Tk], { all: true }),
      3: T("molarity, then moles", [M, mol]),
      4: T("molar mass", [v.mass / mol])
    };
  },

  // Relative rates: divide by its coefficient, multiply by the other
  "relative": function (v) {
    var overall = v.rate / v.coefGiven;
    return {
      4: T("the requested rate", [overall * v.coefWanted], { ignoreSign: true })
    };
  },

  // Method of initial rates
  "rates": function (v) {
    return {
      2: T("order in the first reactant", [v.o1], { exact: true }),
      3: T("order in the second reactant", [v.o2], { exact: true }),
      4: T("rate constant k", [v.k])
    };
  },

  // Integrated rate laws
  "integrated": function (v) {
    var A, half;
    if (v.order === 1) {
      A = v.a0 * Math.exp(-v.k * v.t);
      half = 0.693 / v.k;
    } else if (v.order === 2) {
      A = 1 / (1 / v.a0 + v.k * v.t);
      half = 1 / (v.k * v.a0);
    } else {
      A = v.a0 - v.k * v.t;
      half = v.a0 / (2 * v.k);
    }
    return {
      1: T("reaction order", [v.order], { exact: true }),
      // Instructor, 17 Sept: step 5 is never a calculation. Half-life joins step 4.
      4: T("final concentration or pressure, then half-life", [A, half])
    };
  },

  // Arrhenius — same algebra as Clausius-Clapeyron
  "arrhenius": function (v) {
    var T1 = v.t1C + 273.15, T2 = v.t2C + 273.15;
    // Both temperatures, not either. "23K 293K" cleared this in V1.
    var out = { 2: T("temperatures in Kelvin", [T1, T2], { all: true }) };
    if (v.solveFor === "ea") {
      out[4] = T("activation energy",
                 [R_J * Math.log(v.k2 / v.k1) / (1 / T1 - 1 / T2) / 1000]);
    } else {
      out[4] = T("rate constant at the new temperature",
                 [v.k1 * Math.exp(-(v.ea * 1000 / R_J) * (1 / T2 - 1 / T1))]);
    }
    return out;
  },

  // ICE tables.
  // findK:  rCount = how many reactants each start at a0 (default 2, which is
  //         the H2 + I2 case). N2O4 <=> 2NO2 has rCount 1.
  // perfectSquare: pCoef = product coefficient (default 1). H2 + I2 <=> 2HI
  //         has pCoef 2; CO + H2O <=> CO2 + H2 has pCoef 1.

  "ice": function (v) {
    if (v.mode === "findK") {
      var rCount = (v.rCount === undefined) ? 2 : v.rCount;
      var rCoef = v.rCoef || 1;
      var x = v.prod / v.pCoef;
      var left = v.a0 - rCoef * x;
      var denom = Math.pow(left, rCount * rCoef);
      var Kc = Math.pow(v.prod, v.pCoef) / denom;
      return {
        2: T("the change x", [x]),
        4: icePick(v, { K: Kc, reactant: left }, "equilibrium constant and leftover reactant")
      };
    }
    if (v.mode === "perfectSquare") {
      var r = Math.sqrt(v.K);
      var pc = (v.pCoef === undefined) ? 1 : v.pCoef;
      var rc = v.rCoef || 1;
      // (pc * x) / (a0 - rc * x) = r
      var xs = r * v.a0 / (pc + r * rc);
      return {
        4: icePick(v, { x: xs, product: pc * xs, reactant: v.a0 - rc * xs },
                   "x and the equilibrium concentrations")
      };
    }
    // quadratic: x^2 + Kx - K*a0 = 0
    var xq = (-v.K + Math.sqrt(v.K * v.K + 4 * v.K * v.a0)) / 2;
    return {
      4: icePick(v, { x: xq, product: xq, reactant: v.a0 - xq },
                 "x and the equilibrium concentrations")
    };
  },

  // Q vs K
  "qk": function (v) {
    function prod(list) {
      var p = 1;
      for (var i = 0; i < list.length; i++) p *= Math.pow(list[i][0], list[i][1]);
      return p;
    }
    var Q = prod(v.num) / prod(v.den);
    return {
      3: T("reaction quotient Q", [Q])
      // milestone 4 (compare Q to K, predict the shift) is judged by the tutor:
      // the direction is words, and Q was already verified at milestone 3.
    };
  },

  // ---------- CHAPTER 14 ----------

  // Strong acid / strong base pH and pOH
  // mult = ions released per formula unit (Ba(OH)2 -> 2)
  // want = "pH" or "pOH" when the problem asks for one specific value. Omit it
  //        when the problem asks for pH, pOH and the ion concentration, in
  //        which case milestone 5 accepts either p-value.
  "phpoh": function (v) {
    var c = v.conc * (v.mult || 1);
    var p = -Math.log(c) / Math.LN10;          // p of the primary ion
    var other = 14 - p;
    var otherConc = Math.pow(10, -other);
    var pH  = (v.kind === "acid") ? p : other;
    var pOH = (v.kind === "acid") ? other : p;
    var final;
    if (v.want === "pH") final = [pH];
    else if (v.want === "pOH") final = [pOH];
    else final = [pH, pOH];
    return {
      2: T("primary ion concentration", [c]),
      3: T("the p-value of that ion", [p], { absTol: 0.05 }),
      // Step 5 is never a calculation: the value asked for joins step 4.
      4: T("the other p-value, and the value asked for", [other, otherConc].concat(final),
           { absTols: [0.05, null].concat(final.map(function () { return 0.05; })) })
    };
  },

  // Weak acid or weak base, x-is-small approximation
  "weakacid": function (v) {
    var x = Math.sqrt(v.K * v.c);
    var pH = (v.kind === "base")
      ? 14 + Math.log(x) / Math.LN10
      : -Math.log(x) / Math.LN10;
    // Instructor, 17 Sept: "step 5 should not be a calculation." Percent
    // ionization is part of step 4; step 5 has no number to grade.
    return {
      // Pre-launch run, 17 Sept: with x, pH and percent all accepted singly, the grader
      // cleared on x while the tutor (reading "calculate x, pH and percent") held out
      // for all three, and a wrong 0.95% passed once pH was right. The step is now
      // pH and percent ionization together; x is recognised as progress toward them.
      4: T("pH and percent ionization", [pH, x / v.c * 100], { absTols: [0.05, null], all: true, companions: [x] })
    };
  },

  // Henderson-Hasselbalch.  kIs is "ka" or "kb".
  "buffer": function (v) {
    var pKa = (v.kIs === "kb")
      ? 14 + Math.log(v.K) / Math.LN10
      : -Math.log(v.K) / Math.LN10;
    var ratio = v.base / v.acid;
    return {
      3: T("pKa and the base/acid ratio", [pKa, ratio], { absTol: 0.05, all: true }),
      4: T("pH of the buffer", [pKa + Math.log(ratio) / Math.LN10], { absTol: 0.05 })
    };
  },

  // Titration — three shapes
  "titration": function (v) {
    if (v.mode === "halfEquiv") {
      var pKa = -Math.log(v.Ka) / Math.LN10;
      return { 4: T("pH at half-equivalence", [pKa], { absTol: 0.05 }) };
    }
    if (v.mode === "weakEquiv") {
      var n = v.c * v.V, Vt = v.V * 2, conc = n / Vt;
      var Ka = 1e-14 / v.Kb, x = Math.sqrt(Ka * conc);
      return {
        3: T("moles, then concentration, of the conjugate acid", [n, conc], { milli: true }),
        4: T("pH at the equivalence point", [-Math.log(x) / Math.LN10], { absTol: 0.05 })
      };
    }
    // excess strong acid or base
    var nA = v.cA * v.vA, nB = v.cB * v.vB;
    var ex = Math.abs(nA - nB), Vtot = v.vA + v.vB;
    var c = ex / Vtot;
    var p = -Math.log(c) / Math.LN10;
    return {
      2: T("initial moles of each", [nA, nB], { milli: true, all: true }),
      3: T("moles of the excess species", [ex], { milli: true }),
      4: T("concentration in the new volume, then pH",
           [c, (nA > nB) ? p : 14 - p], { absTols: [null, 0.05] })
    };
  },

  // ---------- CHAPTER 15 ----------

  // Molar solubility in pure water.  stoich 2 = AB (s^2), 3 = AB2 or A2B (4s^3)
  "ksp": function (v) {
    var s = (v.stoich === 3)
      ? Math.pow(v.ksp / 4, 1 / 3)
      : Math.sqrt(v.ksp);
    return { 4: T("molar solubility s", [s]) };
  },

  // Common ion effect
  "commonion": function (v) {
    var s = v.ksp / Math.pow(v.common, v.cPow);
    var pure = (v.stoich === 3) ? Math.pow(v.ksp / 4, 1 / 3) : Math.sqrt(v.ksp);
    return {
      // Step 5 is never a calculation: the pure-water comparison joins step 4.
      4: T("molar solubility with the common ion, then in pure water for comparison", [s, pure, pure / s])
    };
  },

  // Will it precipitate?
  "qksp": function (v) {
    var tot = v.v1 + v.v2;
    var a = v.c1 * v.v1 / tot, b = v.c2 * v.v2 / tot;
    var Q = Math.pow(a, v.p1) * Math.pow(b, v.p2);
    return {
      2: T("diluted ion concentrations", [a, b]),
      4: T("Qsp, compared to Ksp", [Q], { companions: [v.ksp] })
    };
  },

  // Complex ion formation
  "complex": function (v) {
    var c = v.mol / v.L;
    var free = v.nh3 - v.lig * c;
    var Kd = 1 / v.Kf;
    return {
      3: T("Kd", [Kd]),
      4: T("free metal ion concentration", [Kd * c / Math.pow(free, v.lig)]),
      2: T("leftover free ligand", [free])
    };
  },

  // ---------- CHAPTER 16 ----------

  // Reversible entropy: dS = q/T
  "entropy": function (v) {
    var q = v.mol * v.dh * 1000;
    return {
      2: T("temperature in Kelvin", [v.T]),
      4: T("entropy change", [q / v.T], { ignoreSign: true })
    };
  },

  // Gibbs free energy.  mode "dg" or "crossover"
  "gibbs": function (v) {
    if (v.mode === "crossover") {
      return { 3: T("crossover temperature", [v.dh * 1000 / v.ds]) };
    }
    var TdS = v.T * v.ds / 1000;
    var dG = v.dh - TdS;
    var scale = Math.max(Math.abs(v.dh), Math.abs(TdS));
    return {
      2: T("T times delta S in kJ", [TdS], { absTol: 0.02 * scale }),
      3: T("delta G", [dG], { absTol: 0.02 * scale })
    };
  },

  // Non-standard: dG = dG0 + RT ln Q
  "nonstandard": function (v) {
    function prod(list) {
      var p = 1;
      for (var i = 0; i < list.length; i++) p *= Math.pow(list[i][0], list[i][1]);
      return p;
    }
    var Tk = v.T || 298;
    var Q = prod(v.num) / prod(v.den);
    var term = R_J * Tk * Math.log(Q) / 1000;
    var scale = Math.max(Math.abs(v.dg0), Math.abs(term));
    return {
      2: T("reaction quotient Q", [Q]),
      4: T("delta G", [v.dg0 + term], { absTol: 0.02 * scale })
    };
  },

  // dG0 = -RT ln K.  mode "findK" or "findG"
  "keq": function (v) {
    var Tk = v.T || 298;
    if (v.mode === "findG") {
      return { 4: T("standard free energy change", [-R_J * Tk * Math.log(v.K) / 1000]) };
    }
    return { 4: T("equilibrium constant K", [Math.exp(-v.dg0 * 1000 / (R_J * Tk))]) };
  },

  // ---------- CHAPTER 17 ----------

  // Standard cell potential: E = cathode - anode
  "ecell": function (v) {
    var hi = Math.max(v.e1, v.e2), lo = Math.min(v.e1, v.e2);
    return {
      2: T("the two reduction potentials", [v.e1, v.e2], { absTol: 0.02, all: true }),
      3: T("standard cell potential", [hi - lo], { absTol: 0.02 })
    };
  },

  // Nernst: E = E0 - (0.0592/n) log Q
  "nernst": function (v) {
    function prod(list) {
      var p = 1;
      for (var i = 0; i < list.length; i++) p *= Math.pow(list[i][0], list[i][1]);
      return p;
    }
    var Q = prod(v.num) / prod(v.den);
    return {
      2: T("reaction quotient Q", [Q]),
      4: T("cell potential", [v.e0 - (0.0592 / v.n) * Math.log(Q) / Math.LN10],
           { absTol: 0.02 })
    };
  },

  // Faraday's law.  mode "mass" or "time"
  "electrolysis": function (v) {
    var F = 96485;
    if (v.mode === "time") {
      var molMetal = v.mass / v.mm;
      var molE = molMetal * v.n;
      var C = molE * F;
      return {
        1: T("electrons per metal atom", [v.n], { exact: true }),
        3: T("moles of electrons, then total charge", [molE, C]),
        4: T("time", [C / v.current, C / v.current / 60])
      };
    }
    var Q = v.current * v.seconds;
    var e = Q / F;
    var mol = e / v.n;
    return {
      1: T("electrons per metal atom", [v.n], { exact: true }),
      2: T("total charge in coulombs", [Q]),
      3: T("moles of electrons", [e]),
      4: T("mass of metal", [mol * v.mm])
    };
  },

  // ---------- CHAPTER 19 ----------

  // Crystal field: all whole numbers
  "crystal": function (v) {
    return {
      1: T("oxidation state of the metal", [v.ox], { exact: true }),
      2: T("d-electron count", [v.d], { exact: true }),
      4: T("number of unpaired electrons", [v.unpaired], { exact: true })
    };
  },

  // ---------- CHAPTER 21 ----------

  // First-order decay.  Give either a fraction/masses (find time) or a time
  // (find the remaining fraction).
  "decay": function (v) {
    var k = 0.693 / v.half;
    var out = { 2: T("rate constant k", [k, v.half]) };
    if (v.t !== undefined) {
      // Step 5 is never a calculation: the count of half-lives joins step 4.
      out[4] = T("fraction remaining, and the number of half-lives", [Math.exp(-k * v.t), Math.exp(-k * v.t) * 100, v.t / v.half]);
    } else {
      var f = (v.frac !== undefined) ? v.frac : (v.m / v.m0);
      var t = Math.log(1 / f) / k;
      out[4] = T("elapsed time, and the number of half-lives", [t, t / v.half]);
    }
    return out;
  },

  // Mass defect and binding energy
  "binding": function (v) {
    var conv = v.conv || 931.5;
    if (v.defect !== undefined) {
      return { 4: T("total binding energy", [v.defect * conv]) };
    }
    // Atomic masses include electrons. Accept the answer with electron masses
    // added (the consistent method) or without (following bare nucleon masses).
    var A = v.Z + v.N;
    var sum = v.Z * 1.0073 + v.N * 1.0087, sumE = sum + v.Z * 0.00055;
    var d = sum - v.actual, dE = sumE - v.actual;
    var be = d * conv, beE = dE * conv;
    return {
      1: T("sum of the individual particle masses", [sum, sumE]),
      2: T("mass defect", [d, dE]),
      4: T("total binding energy, then per nucleon", [be, be / A, beE, beE / A])
    };
  }
};

function gateStem(w) {
  w = String(w).toLowerCase();
  return w.length > 5 ? w.slice(0, 5) : w;
}

// The words a keyword is satisfied by: itself, its stem, and any synonym group
// it belongs to.
function gateForms(keyword) {
  var w = String(keyword).toLowerCase().replace(/[^a-z\s-]/g, " ").trim();
  var out = [];
  w.split(/[\s-]+/).forEach(function (part) {
    if (!part || GATE_STOP.indexOf(part) > -1) return;
    out.push(gateStem(part));
    for (var g in GATE_SYNONYMS) {
      if (GATE_SYNONYMS[g].some(function (s) { return part.indexOf(s) === 0 || s.indexOf(part) === 0; })) {
        GATE_SYNONYMS[g].forEach(function (s) { out.push(gateStem(s)); });
      }
    }
  });
  return out;
}

var GATE_STOP = ["the", "a", "an", "of", "and", "or", "to", "in", "on", "at", "is", "are", "be", "per", "its", "it", "with", "for"];

// How many of this scenario's keywords the student's own words touch.
function openerCoverage(text, prob) {
  var list = openerKeywords(prob);
  if (!list.length) return { hits: 0, total: 0, ungated: true };
  var t = " " + String(text || "").toLowerCase().replace(/[^a-z0-9\s]/g, " ").replace(/\s+/g, " ") + " ";
  var stems = t.split(" ").map(gateStem);
  var hits = 0;
  list.forEach(function (kw) {
    var forms = gateForms(kw);
    if (forms.some(function (f) { return stems.indexOf(f) > -1 || t.indexOf(" " + f) > -1; })) hits++;
  });
  return { hits: hits, total: list.length, ungated: false };
}

// The keyword list that belongs to the opener this problem is showing.
function openerKeywords(prob) {
  var key = prob && prob.key ? String(prob.key) : "";
  var lists = OPENER_KEYWORDS[key];
  if (!lists) return [];
  var i = openerIndex(prob, Array.isArray(OPENERS[key]) ? OPENERS[key].length : 1);
  var row = Array.isArray(lists) ? lists[i % lists.length] : lists;
  return String(row || "").split(",").map(function (s) { return s.trim(); }).filter(function (s) { return s.length > 2; });
}

// An archetype may have one opener (a string) or several (a list). With several,
// the choice follows the problem variant, so the same problem always gets the same
// opening scenario — on the page, in the prompt, on every turn and after a reload —
// with no state to keep. New Scenario gives a new variant and usually a new opener.
// Which of an archetype's openers this problem shows. Used by openerFor and by
// the scenario check, so the keyword list always belongs to the opener on screen.
function openerIndex(prob, n) {
  if (!n || n < 2) return 0;
  var h = 5381, id = problemId(prob);
  for (var i = 0; i < id.length; i++) h = ((h * 33) ^ id.charCodeAt(i)) >>> 0;
  return h % n;
}

function openerFor(prob) {
  var key = prob && prob.key ? String(prob.key) : "";
  var o = OPENERS[key];
  if (Array.isArray(o) && o.length) return o[openerIndex(prob, o.length)];
  if (o) return o;
  return "First, let us identify the Laws and/or Principles associated with this experiment.";
}


// ---------- 7e. THE MASTERY RECORD ----------
//
// Silent this semester: nothing the student sees changes. It exists so that a
// semester of data accumulates before any of it is attached to a grade, and so
// "did the archetype well" can be calibrated against real students rather than
// guessed at. One row when a problem is finished.
//
// cleanRun means the student reached milestone 5 without ever needing the setup
// handed to them — wrong answers and questions are fine, rung 2 or lower. That is
// the competence condition; milestone 6 is the mastery one built on top of it.
function recordMastery(studentId, prob, sc, milestone) {
  try {
    if (milestone < 1) return;
    // SHEET_ID is declared inside doPost, getMilestone and setMilestone, and
    // nowhere at the top level — so referencing it here threw a ReferenceError on
    // every call, which the catch below swallowed. The sheet was never written and
    // nothing said so. Read the property here, as getMilestone does.
    var ss = SpreadsheetApp.openById(
      PropertiesService.getScriptProperties().getProperty("SHEET_ID"));
    var sheet = ss.getSheetByName("Mastery");
    if (!sheet) {
      sheet = ss.insertSheet("Mastery");
      sheet.appendRow(["Timestamp", "Student ID", "Archetype", "Problem",
                       "Milestone reached", "Deepest scaffold rung", "Turns",
                       "Off-topic strikes", "Clean run", "Outcome"]);
    }
    var worst = sc.worst || 0;
    sheet.appendRow([
      new Date(), studentId || "Guest (No Extra Credit)",
      (prob && prob.key) || "?", problemId(prob),
      milestone, worst, sc.turns || 0, sc.bad || 0,
      (milestone >= 5 && worst <= 2) ? "yes" : "no",
      milestone >= 5 ? "finished" : "abandoned"
    ]);
  } catch (e) {
    // Never let bookkeeping break a student's session — but say so. A silent
    // catch cannot tell "bookkeeping failed" from "bookkeeping never ran".
    try { console.error("recordMastery failed: " + e); } catch (e2) {}
  }
}


// ---------- 7f. DO NOT SAY THE SAME THING EVERY TIME ----------
//
// "Spot on" opened 486 of 1,446 logged replies — a third of everything the tutor
// ever said. Enlarging the list of praise words alone just rotates the tic, so
// the server reads back what has actually been said in this session and names the
// openers that are spent. The model cannot reuse what it is told it has used.
function freshnessNote(history) {
  if (!history || !history.length) return "";
  var used = [], seen = {};
  for (var i = Math.max(0, history.length - 12); i < history.length; i++) {
    var h = history[i];
    if (!h || h.role !== "model" || !h.text) continue;
    // the opening clause, up to the first comma, dash or full stop
    var open = String(h.text).split(/[,.\u2014!?]/)[0].trim();
    if (!open || open.length > 40) continue;
    var key = open.toLowerCase();
    if (seen[key]) continue;
    seen[key] = 1;
    used.push(open);
  }
  if (!used.length) return "";
  return "\n\nALREADY USED IN THIS SESSION, do not open with any of these again: " +
    used.join(" / ") + ". Vary how you acknowledge, and prefer naming the specific thing " +
    "the student got right over praising them for it.";
}

// ---------- 7c. THE SCAFFOLD LADDER ----------
//
// Being stuck is being stuck in ONE account. So escalation moves the student to
// a different register rather than giving a sharper hint in the register they
// are already failing at. Rung 5 is the ceiling: the setup is stated, never
// substituted, and the row is flagged.
//
// Counters live in their own cache key. The milestone cache is the code that
// caused the caching regression; this does not touch it, and adds no sheet reads.
function getScaffold(sessionId, probId) {
  if (!sessionId) return { rung: 0, burn: 0, bad: 0, worst: 0, turns: 0 };
  var raw = CacheService.getScriptCache().get("sc_" + sessionId);
  if (!raw) return { rung: 0, burn: 0, bad: 0, worst: 0, turns: 0 };
  var p = String(raw).split("|");
  if (p[0] !== probId) return { rung: 0, burn: 0, bad: 0, worst: 0, turns: 0 };   // new problem, new ladder
  return { rung: parseInt(p[1], 10) || 0, burn: parseInt(p[2], 10) || 0, bad: parseInt(p[3], 10) || 0,
           worst: parseInt(p[4], 10) || 0, turns: parseInt(p[5], 10) || 0 };
}

function setScaffold(sessionId, probId, sc) {
  if (!sessionId) return;
  CacheService.getScriptCache().put(
    "sc_" + sessionId,
    probId + "|" + sc.rung + "|" + sc.burn + "|" + sc.bad + "|" + sc.worst + "|" + sc.turns, 21600);
}

// advanced: the milestone moved this turn. kind: "question" | "offtopic" | "answer".
function nextScaffold(sc, advanced, kind) {
  var out = { rung: sc.rung, burn: sc.burn, bad: sc.bad,
              worst: sc.worst || 0, turns: (sc.turns || 0) + 1 };
  if (advanced) { out.rung = 0; out.burn = 0; return out; }   // bad strikes persist
  if (kind === "reflection") return out;   // answering a reflection question costs nothing
  if (kind === "offtopic") { out.bad = sc.bad + 1; return out; }
  out.rung = Math.min(5, sc.rung + 1);
  if (out.rung > out.worst) out.worst = out.rung;   // deepest help this problem needed
  // A focused question is the behaviour we want. It buys more help without
  // spending any of the five attempts.
  if (kind !== "question") out.burn = sc.burn + 1;
  return out;
}

// ---------- 5. TURNING A VERDICT INTO PROMPT TEXT ----------

// Quantities that have no units: naming one of these in the label means a bare
// number is a complete answer.
// A label is treated as unitless when it names a pure number and names no
// quantity that carries units. "pH and percent ionization" is unitless; "final
// concentration or pressure, then half-life" is not, because a concentration has
// units even though a count of half-lives does not.
var UNITLESS_PART = /\bpH\b|\bpOH\b|p-value|value asked for|\bpKa\b|\border\b|\bratio\b|percent|fraction|dimensionless|number of|\bcount\b|unpaired|per nucleon|electrons per|compared to|\bQ\b|\bQsp\b|\bK\b|\bKc\b|\bKp\b|\bKsp\b|\bKd\b/;
var DIMENSIONAL_PART = /concentration|molarit|molalit|pressure|temperature|\btime\b|\bmass\b|energy|\bmoles?\b|coulomb|current|potential|voltage|volume|solubilit|rate constant|defect|entropy|\bheat\b/i;
function labelIsUnitless(label) {
  return UNITLESS_PART.test(String(label)) && !DIMENSIONAL_PART.test(String(label));
}
var UNIT_WORDS = /(?:^|[\s\d)])(?:K|°\s*C|degC|J|kJ|mol|mmol|M|m|g|kg|mg|L|mL|atm|mmHg|torr|Pa|s|sec|min|hr|h|V|A|C|nm|cm|kPa|bar|%)(?![A-Za-z])/;

// The student gave the right value with no units on it. Instructor, 18 Sept:
// "My first answer had no units and no tutoring." Units are part of the quantity,
// and rule 11 says so; this makes the server say it rather than hoping.
function unitsNote(g, message) {
  if (!g || !g.label || labelIsUnitless(g.label)) return "";
  if (UNIT_WORDS.test(" " + String(message || ""))) return "";
  return "\nUNITS: their number carries no units. Name only what they actually wrote — do not add the units for them — " +
    "and ask, in one clause, what units that quantity carries. A number only quantifies a unit or a dimension; " +
    "without its units it has no meaning, and on the exam it earns nothing.";
}

// A value accepted only by rounding. When the student gave fewer figures than the
// data supports, a gentle word; otherwise their precision is right and nothing is said.
function roundingNote(g) {
  var r = g && g.rounded;
  if (!r || !r.dataSf || r.sf >= r.dataSf) return "";
  return "\nROUNDING: it is accepted and the milestone may clear, but they gave " + r.sf + " significant figure" +
    (r.sf === 1 ? "" : "s") + " and the data in the problem supports " + r.dataSf + ". In one gentle clause, never as a " +
    "penalty, say that the given data supports " + r.dataSf + " significant figures and invite them to report it that way.";
}

function buildVerdictBlock(g, prob, message, currentMilestone, recent) {
  if (recent !== undefined) message = recent;
  // If the numbers carry the student across several milestones, the ordinary
  // single-milestone verdict is misleading — the last number in the message is
  // their final answer, not their answer to the current step.
  if (prob && message !== undefined && currentMilestone !== undefined) {
    // The tutor kept announcing "now that the problem is on your screen" while the
    // server still had the student at milestone 0 and the panel was still hidden.
    var veilNote = (currentMilestone === 0)
      ? "\n\nThe problem is NOT on the student's screen right now, and this message did not clear " +
        "milestone 1. Do not say the problem is visible, do not refer to \"the problem on your screen\", " +
        "and do not ask about anything in it. You are still on the opening question."
      : "";

    var cap = buildCapNote(prob, message, currentMilestone);
    if (cap) {
      return "\n\nGRADER (server-computed, authoritative — the student cannot see this):" + cap + veilNote;
    }
    var hold = buildHoldNote(prob, message, currentMilestone);
    if (hold) {
      return "\n\nGRADER (server-computed, authoritative — the student cannot see this):" + hold + veilNote;
    }
    // Some archetypes have no numeric anchor anywhere — Le Chatelier is entirely
    // qualitative by design. There the model is the only gate on all five
    // milestones, and it advances on anything that sounds plausible. Say so.
    var unanchored = !(prob && prob.vars && prob.vars.solve);
    if (unanchored && currentMilestone < 5) {
      return "\n\nGRADER (server-computed, authoritative — the student cannot see this):" +
        "\n\nThis problem has NO numeric answer key. Nothing here is checked arithmetically, so you are the " +
        "only gate on every milestone. Hold the bar higher than usual, not lower: a correct label, a correct " +
        "direction, or a good analogy is not enough to advance. Advance only when the student has said what the " +
        "PARTICLES or the competing RATES are doing to produce the effect. When in doubt, ask one more question " +
        "rather than crediting." + veilNote;
    }

    var chain = buildChainNote(prob, message, currentMilestone);
    if (chain) {
      return "\n\nGRADER (server-computed, authoritative — the student cannot see this):" + chain;
    }
  }

  if (!g || g.verdict === "NO_KEY") return "";

  var head = "\n\nGRADER (server-computed, authoritative — the student cannot see this):\n";

  if (g.verdict === "CORRECT") {
    return head +
      "The student's number for " + g.label + " is CORRECT. Confirm it warmly in one sentence, " +
      "then move them to the next milestone. Do not re-derive it." + roundingNote(g) + unitsNote(g, message);
  }
  if (g.verdict === "CORRECT_WITH_STRAY") {
    return head +
      "The student's number for " + g.label + " is CORRECT, but their message also contains the " +
      "value " + g.stray + ", which is not a given in this problem, not a standard constant, and " +
      "not a value this problem produces at any step. Confirm the answer is right, then ask them " +
      "what " + g.stray + " represents and where it came from. Ask, do not assert. They may have a " +
      "good reason. You MAY clear the milestone." + roundingNote(g) + unitsNote(g, message);
  }
  if (g.verdict === "AMBIGUOUS") {
    return head +
      "The student's message contains a correct value for " + g.label + ", but it is not the last " +
      "number they wrote. Ask them to restate just their final answer as a single number. " +
      "Do not say whether it is right or wrong yet. Do NOT clear the milestone.";
  }
  if (g.verdict === "PARTIAL" && g.all) {
    return head +
      "This step needs " + g.needs + " values together: " + g.label + ". Only " + g.student +
      " in the student's message checks out; the rest is missing or wrong. Confirm " + g.student +
      " by naming it, and nothing else in their message. Ask ONE question about what is still missing. " +
      "Do not state any value. Do NOT clear the milestone." + unitsNote(g, message);
  }
  if (g.verdict === "PARTIAL") {
    return head +
      "The student's message contains a correct supporting value (" + g.student + "), but not yet the " +
      "value this milestone needs: " + g.label + ". Acknowledge what is right, then ask ONE question " +
      "toward the missing step. Do not state the value. Do NOT clear the milestone.";
  }
  if (g.verdict === "NO_NUMBER") {
    return head +
      "This milestone requires a numeric answer for " + g.label + " and the student did not give one. " +
      "Ask ONE guiding question toward it. Do NOT clear the milestone.";
  }

  var hint = {
    "SIGN_ERROR":        "The magnitude is right but the sign is wrong. Ask what the sign should mean physically.",
    "CELSIUS_KELVIN":    "The value is off by about 273 — a Celsius/Kelvin mix-up. Ask about the temperature units.",

    "The student's number for " + g.label + " is INCORRECT. " + hint +

}

// ---------- 7. CHAINING MILESTONES ----------
//
// A student who demonstrates several milestones in one message should be
// credited with all of them. The model reports how many it believes were
// demonstrated; this function caps that at what the numbers actually support.
//
// The walk stops at the FIRST milestone that has a numeric target the message
// does not satisfy. Milestones with no numeric target (conceptual steps, or
// problems with no solve block) pass through on the model's judgment, which is
// the same judgment it was already making one step at a time.

// Does any number anywhere in the message satisfy this target? Chaining needs
// this rather than gradeMessage, which deliberately only looks at the last
// number — in a one-milestone-at-a-time turn the last number is the answer, but
// in a full solution the intermediate values are scattered through the message.
// `used` collects the numbers already spent on earlier milestones of this same
// chain. A value the student wrote once cannot answer two different steps: a
// student who writes T1 = 373 K has not thereby also produced T2 = 368 K.
function satisfiesAnywhere(message, target, prob, used) {
  target = withMilli(message, target);
  var nums = extractNumbers(message);
  if (target.all) {
    var got = matchAll(nums, target, used);
    if (got.length < target.values.length) return false;
    if (used) for (var g = 0; g < got.length; g++) used.push(got[g]);
    return true;
  }
  for (var i = 0; i < nums.length; i++) {
    if (used && spentAlready(nums[i], used)) continue;
    if (numberHits(nums[i], target, true)) {
      if (used) used.push(nums[i]);
      return true;
    }
  }
  return false;
}

function spentAlready(x, used) {
  for (var i = 0; i < used.length; i++) {
    if (used[i] === x) return true;
    if (used[i] !== 0 && Math.abs(x - used[i]) <= 1e-9 * Math.abs(used[i])) return true;
  }
  return false;
}


// ---------- 7b. MILESTONE 1 IS AN ACCOUNT, NOT A FORMULA ----------
//
// Measured against 202 real milestone-1 clears in the September log: 143 of
// them were an equation, a bare number, or a one-word fragment. "pi=iMRT",
// "0.085" and "#NAME?" all cleared the submicroscopic gate. This blocks that
// class deterministically. Whether the words actually describe particles is
// still the model's judgment, under rule 11 — a server cannot read meaning.
function hasAccount(message) {
  // Units, variable names and constant symbols are not prose. Without this,
  // "pi = iMRT. pi = 2.54/760 atm ... = 36,600 g/mol" scores four "words".
  var NOT_PROSE = ("atm mol mmol mmoles kg mg ug ng torr mmhg pa kpa bar psi " +
    "kj mj mev amu kelvin celsius min mins hr hrs sec secs " +
    "mm cm km ml dl ppm ppb " +
    "kf kb ka kc kp kw ksp kd keq qsp dh ds dg dt dhvap mm mw " +
    "ph poh pka pkb pi delta lnq " +
    "eq soln aq");
  var stop = {};
  NOT_PROSE.split(" ").forEach(function (w) { if (w) stop[w] = 1; });

  // Prose words must RUN CONSECUTIVELY. A sentence has four in a row; algebra
  // scatters them one at a time between equals signs. Without this, the line
  // "particle M = 7.83e-1/(0.08206 x 298.15) = 3.20e-2 M. solute M = ..." scored
  // "particle", "solute" and "moles" and cleared the submicroscopic gate — and
  // rule 11 tells the student "particles" is the word being looked for.
  var toks = String(message || "").split(/\s+/);
  var run = 0, best = 0;
  for (var i = 0; i < toks.length; i++) {
    var t = toks[i].replace(/[,;:()"'?!]/g, "");
    var ends = /\.$/.test(t);          // a full stop ends the run
    t = t.replace(/\.$/, "");
    var prose = t.length >= 2 &&
      !/[=\u00d7\u00b7^]|\d|10\u207b|E-|e-/.test(t) &&
      /^[A-Za-z\u0394\u03c0\u03c7][A-Za-z\u0394\u03c0\u03c7\u00b0\u207b\u207a\u2080-\u2089\u00b2\u00b3]*$/.test(t) &&
      !stop[t.toLowerCase()];
    if (prose) { run++; if (run > best) best = run; if (ends) run = 0; }
    else run = 0;
  }
  // Three prose words in a row is an account. Four if the message also contains an
  // equation, because a formula with a couple of words around it is not one.
  // "Q will monotonically approach K" is correct and concise, and a flat floor of
  // four rejected it: Q and K are single characters and do not count as words.
  return best >= (/=/.test(String(message || "")) ? 4 : 3);
}

function grantMilestones(prob, message, currentMilestone, claimed, ignoreAccountRule, thisTurnOnly) {
  if (!isFinite(claimed) || claimed < 1) return 0;

  // Milestone 5 is the credit gate. It is where the student has to put the
  // symbolic, submicroscopic and macroscopic accounts together coherently, and
  // it is worth nothing to reach milestone 4 without it. A chain may carry a
  // student up to milestone 4 in one message; milestone 5 is always earned in
  // its own turn. A single ordinary advance into 5 is unaffected.
  // Milestone 1 is the submicroscopic account and milestone 5 is the inference.
  // Both bookends are earned in their own turn; milestones 2 to 4 chain freely.
  // Without this, a single correct number cleared 1 through 4 in one message:
  // henry, relative rates and Ksp have no numeric target before milestone 4, so
  // the walk ran straight through the conceptual steps. 32 such jumps are in
  // the September log.
  // "message" is the recent window: this turn plus the last few. The account may
  // therefore have arrived a turn after the arithmetic, which is exactly what the
  // hold note promises — give the principle and your working is credited.
  // The cap exists only to stop ONE message buying both bookends at once, so it
  // applies only when this turn alone carries an account AND chainable numbers.
  if (currentMilestone === 0 && !ignoreAccountRule) {
    if (!hasAccount(message)) return 0;
    var boughtInOneTurn = (thisTurnOnly === undefined)
      ? true
      : (hasAccount(thisTurnOnly) && grantMilestones(prob, thisTurnOnly, 0, 5, true) >= 2);
    if (boughtInOneTurn && claimed > 1) claimed = 1;
  }

  var maxLanding = (claimed <= 1) ? 5 : 4;
  var room = maxLanding - currentMilestone;
  if (room <= 0) return 0;
  if (claimed > room) claimed = room;

  var reached = 0;        // how far the walk got
  var lastVerified = 0;   // the furthest milestone the NUMBERS actually confirm
  var used = [];          // numbers already spent on an earlier step of this chain

  for (var k = 1; k <= claimed; k++) {
    var ms = currentMilestone + k;
    var t = targetsFor(prob, ms);
    if (t) {
      if (!satisfiesAnywhere(message, t, prob, used)) break;
      lastVerified = k;
    }
    reached = k;
  }

  // A chain is only trusted as far as the arithmetic backs it up. Milestones
  // with no numeric target ride along on the model's judgment, but they cannot
  // carry the chain on their own — otherwise a confident-sounding message with
  // no working in it would advance several steps.
  if (lastVerified === 0) return Math.min(reached, 1);
  return lastVerified;
}

// The last few student turns, newest last, joined into one string. Chaining
// reads this rather than the single latest message: a student who gives the
// value in one turn and its units in the next, or who supplies a later value
// before an earlier one is asked for, has still done the work.
function recentStudentText(history, message) {
  var parts = [];
  if (history && history.length) {
    var turns = history.slice(-6);
    for (var i = 0; i < turns.length; i++) {
      if (turns[i] && turns[i].role === "user" && turns[i].text) parts.push(String(turns[i].text));
    }
  }
  parts.push(String(message || ""));
  return parts.join(" \n ");
}

// Reads the model's reply. Accepts the new integer field and falls back to the
// old boolean so a stale or malformed response still works.
function claimedFromResponse(aiResponse) {
  if (!aiResponse) return 0;
  var n = parseInt(aiResponse.milestones_demonstrated, 10);
  if (isFinite(n) && n >= 0) return n;
  return (aiResponse.milestone_cleared === true) ? 1 : 0;
}

// How far could the numbers in this message carry the student, ignoring what
// the model thinks? Run before the model call so the verdict block can tell it
// a full solution has arrived, rather than reporting the current milestone as
// wrong just because the last number in the message is the final answer.
function previewChain(prob, message, currentMilestone) {
  return grantMilestones(prob, message, currentMilestone, 5 - currentMilestone);
}


// A student who pastes the whole solution as their first message has skipped
// the submicroscopic account, not failed it. The server withholds the credit;
// this tells the tutor to withhold the conversation too. Without it the model
// gets a correct solution and no grader guidance at all, so it congratulates
// the student, spends the milestone 5 question, and closes the problem out —
// while the banner reads 1/5. (Live transcript, first deploy of this change.)
function buildHoldNote(prob, message, currentMilestone) {
  if (currentMilestone !== 0) return "";
  if (hasAccount(message)) return "";
  if (grantMilestones(prob, message, 0, 5, true) < 2) return "";
  return "\n\nThe arithmetic in this message has been checked and much of it is right. " +
    "It is being HELD, not rejected: the student will be credited for every step of it " +
    "as soon as they give the milestone 1 account, and they will not be asked to redo any of it. " +
    "For THIS reply: set milestones_demonstrated to 0. Do NOT confirm their final answer, " +
    "do NOT tell them the arithmetic is correct, do NOT ask the milestone 5 question, and do NOT " +
    "close the problem out. Say in half a sentence that you can see they have run the numbers and " +
    "will come back to them, then ask the milestone 1 question: what are the PARTICLES doing in this " +
    "experiment? One clause about molecules, ions, populations or competing rates is enough.";
}


// The mirror of the hold note. When the account and the arithmetic arrive in the
// SAME message, the server credits milestone 1 and holds the rest — but neither
// the hold note (which needs no account) nor the chain note (which needs a reach
// of 2) fires, so the model used to get a complete correct solution and no
// instruction at all. It then congratulated the student and spent the milestone 5
// question at milestone 1. (Live transcript, relative rates, 12 Sept.)
function buildCapNote(prob, message, currentMilestone) {
  if (currentMilestone !== 0) return "";
  if (!hasAccount(message)) return "";
  var reach = grantMilestones(prob, message, 0, 5, true);
  if (reach < 2) return "";
  return "\n\nThis message contains BOTH the physical account and the arithmetic. Milestone 1 is " +
    "credited now; the rest is verified and HELD, and it will be credited on their next message " +
    "without them repeating any of it. For THIS reply: set milestones_demonstrated to 1. " +
    "Acknowledge the account they gave in half a sentence. Do NOT confirm their final answer, " +
    "do NOT restate their numbers back to them, do NOT tell them the calculation is correct, and " +
    "do NOT ask the milestone 5 question — milestone 5 is four steps away and can only be spent once. " +
    "Ask the milestone " + (currentMilestone + 2) + " question.";
}

function buildChainNote(prob, message, currentMilestone) {
  var reach = previewChain(prob, message, currentMilestone);
  if (reach < 2) return "";
  var landing = currentMilestone + reach;
  return "\n\nThe numbers in this message have been verified against milestones " +
    (currentMilestone + 1) + " through " + landing + " and all of them check out. " +
    "The student has solved these steps in one go. Do NOT walk them back through " +
    "steps they have already shown — acknowledge the whole solution in a sentence " +
    "and set milestones_demonstrated to " + reach + ". Then put the milestone " +
    (landing + 1) + " question to them. That step is the one that matters: they have " +
    "shown the arithmetic, and now they must put the symbolic, particle-level and " +
    "measurable accounts together coherently. Do not accept a restatement of the " +
    "calculation, and do not accept naming a principle without connecting it to the " +
    "quantities in this problem.";
}
